_HWND_BOTTOM = 1
_HWND_TOPMOST = -1
_HWND_NOTOPMOST = -2
_SWP_NOSIZE = 0x0001
_SWP_NOMOVE = 0x0002
_SWP_NOACTIVATE = 0x0010
_SWP_NOOWNERZORDER = 0x0200
_GWL_EXSTYLE = -20
_WS_EX_TOPMOST = 0x00000008
_WS_EX_TRANSPARENT = 0x00000020
_WS_EX_TOOLWINDOW = 0x00000080
_WS_EX_NOACTIVATE = 0x08000000


def _apply_win32_layer(hwnd_int: int, behind_apps: bool) -> None:
    """Keep DarkShadow strictly behind all open Windows apps (never overlapping/blocking them)."""
    if sys.platform != "win32" or not hwnd_int:
        return
    try:
        from ctypes import wintypes

        user32 = ctypes.windll.user32
        user32.SetWindowPos.argtypes = [
            wintypes.HWND,
            wintypes.HWND,
            ctypes.c_int,
            ctypes.c_int,
            ctypes.c_int,
            ctypes.c_int,
            wintypes.UINT,
        ]
        user32.SetWindowPos.restype = wintypes.BOOL

        get_long = getattr(user32, "GetWindowLongPtrW", user32.GetWindowLongW)
        set_long = getattr(user32, "SetWindowLongPtrW", user32.SetWindowLongW)
        get_long.argtypes = [wintypes.HWND, ctypes.c_int]
        get_long.restype = ctypes.c_ssize_t
        set_long.argtypes = [wintypes.HWND, ctypes.c_int, ctypes.c_ssize_t]
        set_long.restype = ctypes.c_ssize_t

        hwnd = ctypes.c_void_p(hwnd_int)
        ex_style = int(get_long(hwnd, _GWL_EXSTYLE))
        new_style = ex_style | _WS_EX_TOOLWINDOW | _WS_EX_NOACTIVATE | _WS_EX_TRANSPARENT
        if new_style != ex_style:
            set_long(hwnd, _GWL_EXSTYLE, new_style)

        flags = _SWP_NOMOVE | _SWP_NOSIZE | _SWP_NOACTIVATE | _SWP_NOOWNERZORDER
        if behind_apps:
            if ex_style & _WS_EX_TOPMOST:
                user32.SetWindowPos(hwnd, ctypes.c_void_p(_HWND_NOTOPMOST), 0, 0, 0, 0, flags)
            user32.SetWindowPos(hwnd, ctypes.c_void_p(_HWND_BOTTOM), 0, 0, 0, 0, flags)
        else:
            user32.SetWindowPos(hwnd, ctypes.c_void_p(_HWND_TOPMOST), 0, 0, 0, 0, flags)
    except Exception:
        pass

        if blocked:

        if self._client is None:
            return "Captain, .env mein GEMINI_API_KEY set kijiye."

        recalled = self.memory.recall_relevant(prompt, limit=4)
        sys_inst = SYSTEM_INSTRUCTION

            sys_inst += "\n\nRECALLED MEMORY FOR CAPTAIN:\n" + "\n".join(f"- {r}" for r in recalled)

        contents: list[Any] = []
        for u, a in self._turns[-6:]:
            contents.append(types.Content(role="user", parts=[types.Part.from_text(text=u)]))
            contents.append(types.Content(role="model", parts=[types.Part.from_text(text=a)]))
        contents.append(types.Content(role="user", parts=[types.Part.from_text(text=prompt)]))

        cfg = types.GenerateContentConfig(
            system_instruction=sys_inst,
            tools=self.mcp.get_live_tools_declarations(non_blocking=False),
        )

        models_to_try = list(dict.fromkeys([self._model, "gemini-4-hydrogen", "gemini-3.1-flash-lite"]))
        last_error: Exception | None = None

        for model_name in models_to_try:
            try:
                convo = list(contents)
                for _step in range(4):
                    resp = self._client.models.generate_content(
                        model=model_name, contents=convo, config=cfg
                    )
                    calls = getattr(resp, "function_calls", None) or []
                    if not calls:
                        text = (resp.text or "").strip()
                        if text:
                            return text
                        break

                    # Preserve exact model Content (including thought_signature parts)
                    if resp.candidates and resp.candidates[0].content:
                        convo.append(resp.candidates[0].content)

                    fn_parts = []
                    for call in calls[:4]:
                        out = self.mcp.execute_tool(str(call.name), dict(call.args or {}))
                        fn_parts.append(
                            types.Part.from_function_response(
                                name=str(call.name),
                                response=out,
                            )
                        )
                    convo.append(types.Content(role="user", parts=fn_parts))
            except Exception as exc:
                last_error = exc
                print(f"[Gemini Fallback] {model_name} failed ({type(exc).__name__}), trying next...")
                continue

        return f"Captain, AI server temporarily busy: {type(last_error).__name__}"

class FastMCP:
    """Standardized MCP & Gemini Non-Blocking Tool Registry with Schema Validation."""

    def __init__(self) -> None:
        self.registry: dict[str, dict[str, Any]] = {}

    def tool(
        self,
        name: str,
        description: str,
        parameters: dict[str, Any] | None = None,
    ) -> Callable:
        def decorator(func: Callable) -> Callable:
            self.registry[name] = {
                "name": name,
                "description": description,
                "parameters": parameters or {"type": "OBJECT", "properties": {}},
                "handler": func,
            }
            return func

        return decorator

    def get_gemini_declarations(self, non_blocking: bool = True) -> list[Any]:
        """Generate google.genai.types.FunctionDeclaration with enum & NON_BLOCKING support."""
        if not HAS_GENAI:
            return []

        decls = []
        for name, meta in self.registry.items():
            props = {}
            raw_props = meta["parameters"].get("properties", {})
            required = meta["parameters"].get("required", [])

            for p_name, p_meta in raw_props.items():
                p_type_str = str(p_meta.get("type", "STRING")).upper()
                p_type = getattr(types.Type, p_type_str, types.Type.STRING)
                schema_kwargs: dict[str, Any] = {
                    "type": p_type,
                    "description": p_meta.get("description", p_name),
                }
                if "enum" in p_meta and p_meta["enum"]:
                    schema_kwargs["enum"] = list(p_meta["enum"])
                props[p_name] = types.Schema(**schema_kwargs)

            schema = types.Schema(
                type=types.Type.OBJECT,
                properties=props,
                required=required if required else None,
            )

            decl_kwargs: dict[str, Any] = {
                "name": name,
                "description": meta["description"],
                "parameters": schema,
            }
            if non_blocking and hasattr(types, "Behavior"):
                decl_kwargs["behavior"] = types.Behavior.NON_BLOCKING

            decls.append(types.FunctionDeclaration(**decl_kwargs))
        return decls

    def get_mcp_manifest(self) -> dict[str, Any]:
        return {
            "protocolVersion": "2024-11-05",
            "tools": [
                {
                    "name": name,
                    "description": meta["description"],
                    "inputSchema": meta["parameters"],
                }
                for name, meta in self.registry.items()
            ],
        }

            parameters={
                "type": "OBJECT",
                "properties": {
                    "action": {
                        "type": "STRING",
                        "enum": ["save", "recall", "remind"],
                        "description": "Memory operation",
                    },
                    "content": {"type": "STRING", "description": "Fact, search query, or reminder message"},
                    "kind": {
                        "type": "STRING",
                        "enum": ["preference", "note"],
                        "description": "Category when saving",
                    },
                    "minutes_from_now": {
                        "type": "INTEGER",
                        "description": "Delay in minutes when action is remind",
                    },
                },
                "required": ["action", "content"],
            },
        )
        def tool_memory_action(
            action: str,
            content: str,
            kind: str = "preference",
            minutes_from_now: int = 10,
        ) -> dict[str, Any]:
            if self._memory is None:
                return {"ok": False, "error_code": "MEMORY_UNAVAILABLE", "message": "Memory engine configured nahi hai."}
            if action == "save":
                ok, msg = self._memory.save_memory(kind, content)
                return {"ok": ok, "verified_state": "memory_saved" if ok else "memory_rejected", "message": msg}
            if action == "recall":
                hits = self._memory.recall_relevant(content, limit=4)
                return {"ok": True, "verified_state": "memory_recalled", "count": len(hits), "memories": hits}
            if action == "remind":
                mins = max(1, int(minutes_from_now or 10))
                due = datetime.now(timezone.utc) + timedelta(minutes=mins)
                ok, msg = self._memory.add_reminder(content, due)
                return {"ok": ok, "verified_state": "reminder_scheduled" if ok else "reminder_rejected", "message": msg}
            return {"ok": False, "error_code": "INVALID_ACTION", "message": f"Unknown memory action '{action}'."}

    def execute_tool(self, name: str, args: dict[str, Any] | None) -> dict[str, Any]:
        """Centralized Tool Execution Middleware (Safety -> Schema Validation -> Execution -> Logging)."""
        meta = self.fast_mcp.registry.get(name)
        if not meta:
            err = {
                "ok": False,
                "error_code": "UNKNOWN_TOOL",
                "message": f"Tool '{name}' registered nahi hai.",
                "available_tools": sorted(self.fast_mcp.registry.keys()),
            }
            log_error("TOOL_ROUTING", err["message"], command_or_context=name)
            return err

        clean_args = dict(args or {})

        # 1. Hard Safety Shield Gate on ALL incoming string arguments
        combined_text = " ".join(str(v) for v in clean_args.values() if isinstance(v, str))
        shield_msg = check_hard_shields(combined_text)
        if shield_msg:
            log_error("SAFETY_SHIELD", shield_msg, command_or_context=f"{name}({clean_args})", status="BLOCKED")
            return {"ok": False, "error_code": "SHIELD_BLOCKED", "message": shield_msg}

        # 2. Strict Required & Enum Schema Validation (enables automatic LLM self-correction)
        params_schema = meta["parameters"]
        props = params_schema.get("properties", {})
        for req_key in params_schema.get("required", []):
            if req_key not in clean_args or clean_args[req_key] is None or clean_args[req_key] == "":
                return {
                    "ok": False,
                    "error_code": "MISSING_ARGUMENT",
                    "message": f"Required parameter '{req_key}' missing for tool '{name}'.",
                    "expected_schema": props.get(req_key, {}),
                }

        for key, val in list(clean_args.items()):
            if key not in props:
                clean_args.pop(key, None)
                continue
            p_meta = props[key]
            allowed_enum = p_meta.get("enum")
            if allowed_enum and val not in allowed_enum:
                return {
                    "ok": False,
                    "error_code": "INVALID_ENUM",
                    "message": f"Invalid value '{val}' for '{key}'.",
                    "allowed_values": allowed_enum,
                }
            # Coerce numeric strings cleanly
            if p_meta.get("type") == "INTEGER" and isinstance(val, str) and val.strip().lstrip("-").isdigit():
                clean_args[key] = int(val.strip())
            elif p_meta.get("type") == "NUMBER" and isinstance(val, (int, float, str)):
                try:
                    clean_args[key] = float(val)
                except Exception:
                    pass

        # 3. Execute handler with structured error capture & Antigravity error logging
        handler = meta["handler"]
        try:
            result = handler(**clean_args)
            if isinstance(result, dict):
                return result
            return {"ok": True, "output": str(result)}
        except Exception as exc:
            log_error(
                category="TOOL_EXECUTION",
                error_message=f"{type(exc).__name__}: {exc}",
                command_or_context=f"{name}({clean_args})",
                exception=exc,
            )
            return {
                "ok": False,
                "error_code": "EXECUTION_ERROR",
                "message": f"Tool '{name}' failed: {type(exc).__name__}: {exc}",
            }

def compute_fft_bands(pcm_int16: np.ndarray, sample_rate: int = 16000) -> tuple[float, float, float]:
    """Calculate normalized (bass, mid, treble) bands [0..1] for 3D particle audio breathing."""
    if pcm_int16.size < 64:
        return 0.0, 0.0, 0.0
    samples = pcm_int16.astype(np.float32) / 32768.0
    windowed = samples * np.hanning(samples.size)
    spectrum = np.abs(np.fft.rfft(windowed))
    freqs = np.fft.rfftfreq(samples.size, d=1.0 / sample_rate)

    bass = float(np.mean(spectrum[(freqs >= 60) & (freqs < 280)])) if np.any((freqs >= 60) & (freqs < 280)) else 0.0
    mid = float(np.mean(spectrum[(freqs >= 280) & (freqs < 2200)])) if np.any((freqs >= 280) & (freqs < 2200)) else 0.0
    treble = float(np.mean(spectrum[(freqs >= 2200) & (freqs <= 7500)])) if np.any((freqs >= 2200) & (freqs <= 7500)) else 0.0

    return (
        min(1.0, bass * 0.28),
        min(1.0, mid * 0.38),
        min(1.0, treble * 0.60),
    )

    def _speaker_worker(self) -> None:
        """Plays 24kHz mono PCM audio chunks streamed from Gemini Live."""
        try:
            with sd.RawOutputStream(
                samplerate=SPEAKER_RATE,
                channels=1,
                dtype="int16",
                blocksize=1920,
            ) as stream:
                while not self._stop_event.is_set():
                    try:
                        chunk = self._speaker_queue.get(timeout=0.06)
                    except queue.Empty:
                        if self._is_speaking and (time.time() - self._last_spoke_time > 0.65):
                            self._is_speaking = False
                            self._on_status("listening" if self.active else "standby")
                            self._on_fft(0.0, 0.0, 0.0)
                        continue

                    if self._stop_event.is_set():
                        break

                    if not self._is_speaking:
                        self._is_speaking = True
                        self._on_status("speaking")
                    self._last_spoke_time = time.time()
                    stream.write(chunk)
                    arr = np.frombuffer(chunk, dtype=np.int16)
                    b, m, tr = compute_fft_bands(arr, SPEAKER_RATE)
                    self._on_fft(b, m, tr)
        except Exception as exc:
            log_error("SPEAKER_WORKER", str(exc), exception=exc)
            print(f"[Speaker Worker Error] {exc}")

    async def _run_tool_async(self, session: Any, fc: Any) -> None:
        """Execute a tool off the event loop with timeout and send scheduled FunctionResponse."""
        self._on_status("processing")
        args_dict = dict(fc.args or {})
        print(f"[Live Tool Call] {fc.name}({args_dict})")

            tool_res = await asyncio.wait_for(
                asyncio.to_thread(self._tool_executor.execute_tool, fc.name, args_dict),
                timeout=15.0,

                "ok": False,

                "message": f"Tool '{fc.name}' timed out after 15 seconds.",

        except Exception as exc:
            tool_res = {
                "ok": False,
                "error_code": "ASYNC_DISPATCH_ERROR",
                "message": f"{type(exc).__name__}: {exc}",
            }
            log_error("TOOL_ASYNC", str(exc), command_or_context=f"{fc.name}", exception=exc)

        print(f"[Live Tool Result] {fc.name} -> {tool_res}")

        response_payload = tool_res if isinstance(tool_res, dict) else {"ok": True, "output": str(tool_res)}

        # Choose smart scheduling: urgent shield blocks interrupt; quiet memory saves stay non-intrusive
        fr_kwargs: dict[str, Any] = {
            "name": fc.name,
            "id": fc.id,
            "response": response_payload,

        if hasattr(types, "FunctionResponseScheduling"):
            if response_payload.get("error_code") == "SHIELD_BLOCKED":
                fr_kwargs["scheduling"] = types.FunctionResponseScheduling.INTERRUPT
                self._on_status("alert")
            else:
                fr_kwargs["scheduling"] = types.FunctionResponseScheduling.WHEN_IDLE

        try:

                function_responses=[types.FunctionResponse(**fr_kwargs)]
            )
        except Exception as exc:
            log_error("TOOL_RESPONSE_SEND", str(exc), command_or_context=fc.name, exception=exc)

    def _run_async_loop(self) -> None:
        self._loop = asyncio.new_event_loop()
        asyncio.set_event_loop(self._loop)
        self._loop.run_until_complete(self._live_session_loop())

    async def _live_session_loop(self) -> None:
        self._mic_queue = asyncio.Queue(maxsize=150)
        loop = asyncio.get_running_loop()

        def enqueue_mic_safe(raw: bytes) -> None:
            if self._mic_queue is None:
                return
            if self._mic_queue.full():
                try:
                    self._mic_queue.get_nowait()
                except asyncio.QueueEmpty:
                    pass
            try:
                self._mic_queue.put_nowait(raw)
            except asyncio.QueueFull:
                pass

        def mic_callback(indata, _frames, _time_info, _status):
            if self._stop_event.is_set():
                return

            # Strict Acoustic Echo Gate:
            # Drop mic audio completely while assistant is speaking or within 0.70s after speech stops.
            # This completely prevents laptop speakers from echoing into the mic and causing self-hallucination loops!
            if self._is_speaking or (time.time() - self._last_spoke_time < 0.70):
                return

            raw = bytes(indata)
            decision = self._voice_pipeline.process_pcm16_frame(
                raw, assistant_is_speaking=False
            )

            arr = np.frombuffer(raw, dtype=np.int16)

            if self.active:
                loop.call_soon_threadsafe(enqueue_mic_safe, raw)
                if decision.rms > 0.0015:
                    b, m, tr = compute_fft_bands(arr, MIC_RATE)
                    self._on_fft(b, m, tr)
            else:
                if decision.is_speech:
                    self._standby_buffer.append(raw)
                    self._standby_speech_active = True
                    self._standby_speech_chunks += 1
                    b, m, tr = compute_fft_bands(arr, MIC_RATE)
                    self._on_fft(b * 0.35, m * 0.35, tr * 0.35)
                elif self._standby_speech_active:
                    self._standby_buffer.append(raw)
                    self._standby_speech_chunks += 1
                    if self._standby_speech_chunks >= 12:
                        full_pcm = b"".join(self._standby_buffer)
                        self._standby_buffer.clear()
                        self._standby_speech_active = False
                        self._standby_speech_chunks = 0
                        self._process_standby_audio_for_wakeword(full_pcm)

        active_model = self._primary_model

        while not self._stop_event.is_set():
            use_ext = "extended-thinking" in active_model
            cfg = self._build_live_config(use_extended_thinking=use_ext)
            print(f"[Gemini Live] Connecting WebSocket ({active_model})...")
            self._on_status("processing")

            try:
                async with self._client.aio.live.connect(model=active_model, config=cfg) as session:
                    self._current_session = session
                    print(f"[Gemini Live] Connected ({active_model})!")
                    self._on_status("listening" if self.active else "standby")

                    if not self._greeting_sent:
                        self._greeting_sent = True
                        self._on_caption("हाँ कैप्टन, DarkShadow Live Core ऑनलाइन है।", "DARKSHADOW LIVE")
                        await session.send_client_content(
                            turns=types.Content(
                                role="user",
                                parts=[
                                    types.Part.from_text(
                                        text="Captain Kuldeep just launched DarkShadow. Greet Captain in one crisp, warm Hinglish sentence."
                                    )
                                ],
                            ),
                            turn_complete=True,
                        )

                    async def mic_streamer():
                        with sd.RawInputStream(
                            samplerate=MIC_RATE,
                            channels=1,
                            dtype="int16",
                            blocksize=MIC_BLOCK,
                            callback=mic_callback,
                        ):
                            while not self._stop_event.is_set():
                                try:
                                    pcm = await asyncio.wait_for(self._mic_queue.get(), timeout=0.1)
                                    if self.active:
                                        await session.send_realtime_input(
                                            audio={"data": pcm, "mime_type": "audio/pcm;rate=16000"}
                                        )
                                except asyncio.TimeoutError:
                                    continue
                                except Exception:
                                    break

                    mic_task = asyncio.create_task(mic_streamer())
                    pending_tool_tasks: set[asyncio.Task] = set()

                    try:
                        # Public continuous receive loop across multiple turns
                        while not self._stop_event.is_set():
                            async for msg in session.receive():
                                if self._stop_event.is_set():
                                    break

                                # Non-blocking Tool Dispatch (NEVER blocks WebSocket receive!)
                                if msg.tool_call and self._tool_executor is not None:
                                    for fc in msg.tool_call.function_calls:
                                        task = asyncio.create_task(self._run_tool_async(session, fc))
                                        pending_tool_tasks.add(task)
                                        task.add_done_callback(pending_tool_tasks.discard)
                                    continue

                                sc = msg.server_content
                                if sc is None:
                                    continue

                                if sc.interrupted:
                                    self._clear_speaker_queue()
                                    continue

                                # Live Audio Transcriptions -> Captions, Semantic Turn & Living Memory
                                in_tr = getattr(sc, "input_transcription", None)
                                if in_tr and getattr(in_tr, "text", None):
                                    self._current_user_transcript.append(in_tr.text)
                                    joined_user = "".join(self._current_user_transcript).strip()
                                    self._voice_pipeline.update_partial_transcript(joined_user)
                                    self._on_caption(joined_user, "CAPTAIN")

                                out_tr = getattr(sc, "output_transcription", None)
                                if out_tr and getattr(out_tr, "text", None):
                                    self._current_model_transcript.append(out_tr.text)
                                    self._on_caption("".join(self._current_model_transcript).strip(), "DARKSHADOW")

                                model_turn = sc.model_turn
                                if model_turn is not None:

                                            self._enqueue_speaker_nonblocking(part.inline_data.data)

                                is_idle = (
                                    str(interaction_st).endswith("IDLE")
                                    if interaction_st is not None
                                    else bool(getattr(sc, "turn_complete", False))
                                )
                                if is_idle and (self._current_user_transcript or self._current_model_transcript):
                                    u_txt = "".join(self._current_user_transcript).strip()
                                    m_txt = "".join(self._current_model_transcript).strip()
                                    self._current_user_transcript.clear()
                                    self._current_model_transcript.clear()

                                            target=LIVING_MEMORY.ingest_conversation_turn,

                                            daemon=True,
                                        ).start()
                                        if self._on_turn_complete:
                                            self._on_turn_complete(u_txt, m_txt)
                    finally:
                        mic_task.cancel()
                        for t_task in list(pending_tool_tasks):
                            t_task.cancel()
                        self._current_session = None

            except Exception as exc:
                err_str = str(exc)
                print(f"[Gemini Live Reconnect] {type(exc).__name__}: {err_str}")
                log_error("LIVE_WEBSOCKET", err_str, command_or_context=active_model, exception=exc)
                # Automatic model fallback if extended-thinking is temporarily unavailable
                if active_model != self._fallback_model and any(
                    k in err_str.lower() for k in ("404", "not found", "unsupported", "503")
                ):
                    print(f"[Gemini Live] Falling back to {self._fallback_model}")
                    active_model = self._fallback_model
                self._current_session = None
                self._on_status("alert")
                await asyncio.sleep(2.0)

_EXTRACT_ELEMENTS_JS = """
() => {
  const selectors = 'a[href], button, input, textarea, [role="button"], [role="link"], [role="textbox"], [contenteditable="true"]';
  const nodes = Array.from(document.querySelectorAll(selectors));
  const visible = [];
  let idx = 1;
  for (const el of nodes) {
    const rect = el.getBoundingClientRect();
    if (rect.width < 4 || rect.height < 4) continue;
    if (rect.bottom < 0 || rect.top > window.innerHeight * 1.4) continue;
    const style = window.getComputedStyle(el);
    if (style.visibility === 'hidden' || style.display === 'none' || style.opacity === '0') continue;

    const label = (
      el.getAttribute('aria-label') ||
      el.getAttribute('placeholder') ||
      el.getAttribute('title') ||
      el.innerText ||
      el.value ||
      ''
    ).replace(/\\s+/g, ' ').trim().slice(0, 90);

    if (!label && el.tagName !== 'INPUT' && el.tagName !== 'TEXTAREA') continue;

    el.setAttribute('data-ds-idx', String(idx));
    visible.push({
      id: idx,
      tag: el.tagName.toLowerCase(),
      role: el.getAttribute('role') || el.getAttribute('type') || '',
      label: label || '(input field)'
    });
    idx++;
    if (idx > 45) break;
  }
  return {
    title: document.title,
    url: location.href,
    elements: visible
  };
}
"""


def _is_cdp_reachable(cdp_url: str = CDP_URL) -> bool:
    try:
        with urllib.request.urlopen(f"{cdp_url}/json/version", timeout=0.8) as resp:
            return resp.status == 200
    except Exception:
        return False


def ensure_edge_cdp(cdp_url: str = CDP_URL) -> bool:
    """Return True if Microsoft Edge CDP is reachable; otherwise launch Edge with debugging port."""
    if _is_cdp_reachable(cdp_url):
        return True
    try:
        port = cdp_url.rsplit(":", 1)[-1]
        user_data = os.path.expandvars(r"%LOCALAPPDATA%\DarkShadowEdgeProfile")
        subprocess.Popen(
            [
                "cmd",
                "/c",
                "start",
                "",
                "msedge",
                f"--remote-debugging-port={port}",
                f"--user-data-dir={user_data}",
                "--no-first-run",
            ],
            shell=False,
        )
        for _ in range(10):
            time.sleep(0.35)
            if _is_cdp_reachable(cdp_url):
                return True
    except Exception:
        pass
    return False


class EdgeBrowserAgent:
    """Playwright CDP agent that controls Captain's live Microsoft Edge tabs safely."""

    def __init__(self, cdp_url: str = CDP_URL) -> None:
        self.cdp_url = cdp_url
        self.last_youtube_url: str = ""
        self.last_video_title: str = ""

    def _run_on_active_page(self, fn):
        if not HAS_PLAYWRIGHT:
            return None, "Playwright installed nahi hai (pip install playwright)."
        if not _is_cdp_reachable(self.cdp_url):
            if not ensure_edge_cdp(self.cdp_url):
                return None, "Edge CDP port 9222 unreachable. Run Launch_Edge_Debug.bat once."
        try:
            with sync_playwright() as p:
                browser = p.chromium.connect_over_cdp(self.cdp_url, timeout=5000)
                contexts = browser.contexts
                ctx = contexts[0] if contexts else browser.new_context()
                pages = ctx.pages
                page = pages[-1] if pages else ctx.new_page()
                result = fn(page, ctx)
                # Do NOT close Captain's live Edge browser; just disconnect cleanly!
                return result, None
        except Exception as exc:
            return None, f"CDP automation error: {type(exc).__name__}: {exc}"

    def list_interactive_elements(self) -> dict[str, Any]:
        """Snapshot numbered clickable/typeable elements on the active Edge tab."""
        data, err = self._run_on_active_page(lambda page, _ctx: page.evaluate(_EXTRACT_ELEMENTS_JS))
        if err:
            return {"ok": False, "error_code": "CDP_UNAVAILABLE", "message": err}
        return {"ok": True, "verified_state": "dom_snapshot_ready", **data}

    def read_active_page(self, url: str = "") -> dict[str, Any]:
        """Navigate (optional) and extract clean title + readable text from Edge."""
        target = url.strip()
        if target and not target.startswith(("http://", "https://")):
            target = f"https://{target}"

        def _op(page, _ctx):
            if target:
                page.goto(target, timeout=12000, wait_until="domcontentloaded")
            title = page.title()
            text = page.inner_text("body")[:1400].strip()
            return {"title": title, "url": page.url, "preview": text}

        data, err = self._run_on_active_page(_op)
        if err:
            if target:
                fallback = open_in_edge(target)
                fallback["note"] = f"Opened in Edge without text extraction ({err})."
                return fallback
            return {"ok": False, "error_code": "CDP_UNAVAILABLE", "message": err}
        return {"ok": True, "verified_state": "page_read", **data}

    def navigate_active_tab(self, url: str) -> dict[str, Any]:

    def click_element(self, element_id: int | None = None, text_query: str = "") -> dict[str, Any]:
        """Click an element by its data-ds-idx number OR visible text/aria-label with payment guard."""
        if text_query:
            shield = check_hard_shields(text_query)
            if shield:
                return {"ok": False, "error_code": "SHIELD_BLOCKED", "message": shield}

        def _op(page, _ctx):
            if element_id is not None:
                loc = page.locator(f'[data-ds-idx="{int(element_id)}"]').first
                label = (loc.inner_text(timeout=1500) or loc.get_attribute("aria-label") or "").strip()
                shield_el = check_hard_shields(label)

                    return {"blocked": shield_el}
                loc.click(timeout=4000)

                return {"clicked": label or f"#{element_id}", "url": page.url, "title": page.title()}

            if text_query:

                loc.click(timeout=4000)
                time.sleep(0.4)
                return {"clicked": text_query, "url": page.url, "title": page.title()}

            return {"error": "Provide element_id or text_query."}

        data, err = self._run_on_active_page(_op)
        if err:
            return {"ok": False, "error_code": "CLICK_FAILED", "message": err}
        if isinstance(data, dict) and data.get("blocked"):

        if isinstance(data, dict) and data.get("error"):
            return {"ok": False, "error_code": "INVALID_ARGS", "message": data["error"]}
        return {
            "ok": True,
            "verified_state": "element_clicked",
            **data,
            "message": f"'{data.get('clicked')}' par click kar diya hai.",
        }

    def type_and_submit(self, text: str, element_id: int | None = None, press_enter: bool = True) -> dict[str, Any]:

            **data,
            "message": f"Input field mein '{text[:60]}' type{' aur submit' if press_enter else ''} kar diya.",
        }

    def scroll_page(self, direction: str = "down", pixels: int = 550) -> dict[str, Any]:

        if err:

        return {"ok": True, "verified_state": "scrolled", "scroll_y": data, "message": f"Page {direction} scroll kar diya."}

    def play_media_in_edge(self, query: str, platform: str = "youtube", start_at_seconds: float = 0.0) -> dict[str, Any]:

def check_hard_shields(text: str) -> str | None:
    """Unbypassable deterministic safety check applied to every command and tool payload."""
    if not text:
        return None
    if PAYMENT_RE.search(text):
        return (
            "Security Shield: Automated payment, checkout aur financial credentials "
            "strictly blocked hain. Final payment kripya manually complete kijiye."
        )
    if CAMERA_RE.search(text):
        return (
            "Privacy Shield: Physical camera/webcam hardware access permanently disabled hai. "
            "Sirf on-demand monitor screen vision active hai."
        )
    if DANGEROUS_SHELL_RE.search(text):
        return "Security Shield: Destructive system ya disk command strictly blocked hai."
    return None


def get_active_window_title() -> str:
    """Return the title of the currently focused Windows window for active context awareness."""
    if not HAS_USER32:
        return ""
    try:
        hwnd = _user32.GetForegroundWindow()
        if not hwnd:
            return ""
        length = _user32.GetWindowTextLengthW(hwnd)
        if length <= 0:
            return ""
        buf = ctypes.create_unicode_buffer(length + 1)
        _user32.GetWindowTextW(hwnd, buf, length + 1)
        return buf.value.strip()
    except Exception:
        return ""

def close_application(
    app_name: str, on_close_callback: Callable[[], None] | None = None
) -> dict[str, Any]:
    """Close one allowlisted app or active window cleanly and verify termination."""
    clean = str(app_name).strip().lower()
    if not clean:
        return {"ok": False, "error_code": "INVALID_ARGS", "message": "app_name specify kijiye."}

    if clean in ("darkshadow", "dark shadow", "this assistant", "khud ko"):
        if on_close_callback:
            threading.Timer(1.0, on_close_callback).start()
            return {
                "ok": True,
                "verified_state": "shutdown_scheduled",
                "message": "DarkShadow band ho raha hai, Captain. Alvida!",
            }
        return {"ok": True, "verified_state": "shutdown_requested", "message": "Shutdown requested."}

    if clean in ("active", "current", "active window", "this window", "window"):
        if HAS_USER32:
            hwnd = _user32.GetForegroundWindow()
            if hwnd:
                title = get_active_window_title()
                _user32.PostMessageW(hwnd, 0x0010, 0, 0)
                return {
                    "ok": True,
                    "verified_state": "wm_close_sent",
                    "window": title,
                    "message": f"Active window '{title or 'current'}' ko close signal bhej diya.",
                }
        return {"ok": False, "error_code": "NO_ACTIVE_WINDOW", "message": "Koi active window nahi mili."}

    target_procs = APPLICATION_PROCESS_NAMES.get(clean)
    if not target_procs:
        for k, v in APPLICATION_PROCESS_NAMES.items():
            if k in clean or clean in k:
                target_procs = v
                break

    if not target_procs:
        return {
            "ok": False,
            "error_code": "APP_NOT_FOUND",
            "message": f"'{app_name}' supported close list mein nahi mila.",
            "allowed_values": sorted(APPLICATION_PROCESS_NAMES.keys()),
        }

    closed_any = False
    for proc_name in target_procs:
        try:
            res = subprocess.run(
                ["taskkill", "/IM", proc_name, "/F", "/T"],
                capture_output=True,
                text=True,
                check=False,
                creationflags=getattr(subprocess, "CREATE_NO_WINDOW", 0),
            )
            if res.returncode == 0:
                closed_any = True
        except Exception:
            pass

    if closed_any:
        return {
            "ok": True,
            "verified_state": "process_terminated",
            "app": clean,
            "message": f"'{app_name}' band kar diya gaya hai, Captain.",
        }
    return {
        "ok": False,
        "error_code": "NOT_RUNNING",
        "message": f"'{app_name}' abhi running nahi mila.",
    }


def set_system_volume(
    level: int | None = None,
    direction: str = "",
    amount: int = 10,
    mute: bool | None = None,
) -> dict[str, Any]:
    """Set Windows master volume or mute state with verified feedback."""
    clean_dir = str(direction).strip().lower()

    if HAS_PYCAW:
        try:
            speakers = AudioUtilities.GetSpeakers()
            endpoint = speakers.EndpointVolume
            if mute is not None or clean_dir in ("mute", "unmute"):
                is_mute = True if (mute is True or clean_dir == "mute") else False
                endpoint.SetMute(1 if is_mute else 0, None)
                return {
                    "ok": True,
                    "verified_state": "muted" if is_mute else "unmuted",
                    "message": "Audio mute kar diya." if is_mute else "Audio unmute kar diya.",
                }
            if level is not None:
                lvl = max(0, min(100, int(level)))
                endpoint.SetMasterVolumeLevelScalar(lvl / 100.0, None)
                return {
                    "ok": True,
                    "verified_state": "volume_set",
                    "volume_percent": lvl,
                    "message": f"Master volume {lvl}% set kar diya.",
                }
            if clean_dir in ("up", "down"):
                current = round(float(endpoint.GetMasterVolumeLevelScalar()) * 100)
                step = max(1, min(50, int(amount)))
                target = max(0, min(100, current + (step if clean_dir == "up" else -step)))
                endpoint.SetMasterVolumeLevelScalar(target / 100.0, None)
                return {
                    "ok": True,
                    "verified_state": "volume_adjusted",
                    "volume_percent": target,
                    "message": f"Volume {target}% kar diya.",
                }
        except Exception:
            pass

    vk = None
    if clean_dir == "up":
        vk = 0xAF
    elif clean_dir == "down":
        vk = 0xAE
    elif clean_dir in ("mute", "unmute") or mute is not None:
        vk = 0xAD

    if vk and HAS_USER32:
        steps = 1 if vk == 0xAD else max(1, min(20, amount // 2))
        for _ in range(steps):
            _user32.keybd_event(vk, 0, 0, 0)
            _user32.keybd_event(vk, 0, 2, 0)
        return {"ok": True, "verified_state": "win32_key_sent", "message": f"Volume '{clean_dir}' execute hua."}

    return {"ok": False, "error_code": "AUDIO_UNAVAILABLE", "message": "Volume control unavailable."}

def control_active_window(action: str) -> dict[str, Any]:
    """Control active window or Windows desktop state."""
    act = str(action).strip().lower()
    if act == "show_desktop":
        if HAS_USER32:
            _user32.keybd_event(0x5B, 0, 0, 0)
            _user32.keybd_event(ord("D"), 0, 0, 0)
            time.sleep(0.03)
            _user32.keybd_event(ord("D"), 0, 2, 0)
            _user32.keybd_event(0x5B, 0, 2, 0)
            return {"ok": True, "verified_state": "desktop_toggled", "message": "Show Desktop toggle kar diya."}
        if HAS_PYAUTOGUI:
            pyautogui.hotkey("win", "d")
            return {"ok": True, "verified_state": "desktop_toggled", "message": "Show Desktop toggle kar diya."}

    if act == "lock_pc":
        if HAS_USER32:
            _user32.LockWorkStation()
            return {"ok": True, "verified_state": "workstation_locked", "message": "PC lock kar diya hai."}
        subprocess.Popen(["rundll32.exe", "user32.dll,LockWorkStation"], shell=False)
        return {"ok": True, "verified_state": "workstation_locked", "message": "PC lock kar diya hai."}

    if HAS_USER32:
        hwnd = _user32.GetForegroundWindow()
        if hwnd:
            codes = {"minimize": 6, "maximize": 3, "restore": 9}
            if act in codes:
                _user32.ShowWindow(hwnd, codes[act])
                return {"ok": True, "verified_state": f"window_{act}", "message": f"Active window {act} kar diya."}
            if act == "close":
                _user32.PostMessageW(hwnd, 0x0010, 0, 0)
                return {"ok": True, "verified_state": "window_closed", "message": "Active window close kar diya."}

    return {"ok": False, "error_code": "WINDOW_ACTION_FAILED", "message": f"Window action '{act}' fail hua."}


def resolve_contact(query: str) -> dict[str, Any] | None:
    """Resolve contact name or alias from Captain's local contacts.json."""
    candidate_paths = [
        Path(__file__).resolve().parent.parent / "contacts.json",
        Path(__file__).resolve().parent / "contacts.json",
        Path.cwd() / "contacts.json",
    ]
    contacts_file = next((p for p in candidate_paths if p.exists()), None)
    if not contacts_file:
        return None
    try:
        with open(contacts_file, encoding="utf-8") as f:
            data = json.load(f)
        q = query.strip().lower()
        for c in data:
            if c.get("name", "").lower() == q or q in [str(a).lower() for a in c.get("aliases", [])]:
                return c
        for c in data:
            if q in c.get("name", "").lower():
                return c
    except Exception:
        pass
    return None


def send_whatsapp_message(recipient: str, message: str) -> dict[str, Any]:
    """Open WhatsApp desktop/Edge Web with pre-filled message for a contact or phone number."""
    rec = str(recipient).strip()
    msg = str(message).strip()
    if not rec or not msg:
        return {"ok": False, "error_code": "INVALID_ARGS", "message": "Recipient aur message dono chahiye."}

    shield_err = check_hard_shields(msg)
    if shield_err:
        return {"ok": False, "error_code": "SHIELD_BLOCKED", "message": shield_err}

    phone = ""
    contact = resolve_contact(rec)
    if contact and contact.get("phone"):
        phone = str(contact["phone"])
    else:
        cleaned = re.sub(r"[^\d+]", "", rec)
        if len(cleaned) >= 7:
            phone = cleaned

    if phone:
        digits = re.sub(r"[^\d]", "", phone)
        if len(digits) == 10 and not phone.startswith("+"):
            digits = "91" + digits
        encoded_msg = urllib.parse.quote(msg)
        try:
            subprocess.Popen(
                ["cmd", "/c", "start", "", f"whatsapp://send?phone={digits}&text={encoded_msg}"],
                shell=False,
            )
            return {
                "ok": True,
                "verified_state": "whatsapp_draft_opened",
                "recipient": rec,
                "phone": digits,
                "message": f"WhatsApp mein {rec} ke liye message draft open kar diya hai. Enter dabakar send kar sakte hain.",
            }
        except Exception:
            return open_in_edge(f"https://web.whatsapp.com/send?phone={digits}&text={encoded_msg}")

    return {
        "ok": False,
        "error_code": "CONTACT_NOT_FOUND",
        "message": f"Contact '{rec}' contacts.json mein nahi mila aur phone number bhi nahi diya gaya.",
        "hint": "Captain se phone number poochiye ya WhatsApp Desktop open karke screen inspect kijiye.",
    }


def search_local_files(query: str, folder: str = "", max_results: int = 8) -> dict[str, Any]:
    """Bounded local file search (capped at 4 seconds so it never hangs the assistant)."""
    q = str(query).strip().lower()
    if not q:
        return {"ok": False, "error_code": "INVALID_ARGS", "message": "Search query empty hai."}

    folder_map = {
        "downloads": Path.home() / "Downloads",
        "documents": Path.home() / "Documents",
        "desktop": Path.home() / "Desktop",
    }
    roots: list[Path] = []
    if folder:
        mapped = folder_map.get(folder.strip().lower())
        if mapped:
            roots.append(mapped)
        elif os.path.exists(folder):
            roots.append(Path(folder))
    if not roots:
        roots = list(folder_map.values())

    found: list[str] = []
    deadline = time.monotonic() + 4.0
    for root in roots:
        if not root.exists():
            continue
        for dirpath, dirnames, filenames in os.walk(root):
            if time.monotonic() > deadline or len(found) >= max_results:
                break
            # Skip heavy hidden/build directories
            dirnames[:] = [
                d for d in dirnames
                if d not in ("node_modules", ".git", ".venv", "__pycache__", "AppData")
            ]
            for fname in filenames:
                if q in fname.lower():
                    found.append(str(Path(dirpath) / fname))
                    if len(found) >= max_results:
                        break
        if time.monotonic() > deadline or len(found) >= max_results:
            break

    if found:
        return {"ok": True, "verified_state": "files_found", "count": len(found), "files": found}
    return {"ok": True, "verified_state": "no_matches", "count": 0, "files": [], "message": f"'{query}' naam ki koi file nahi mili."}

def capture_monitor_jpeg_bytes() -> bytes | None:
    """Capture primary monitor snapshot on-demand with multi-tier fallback (Zero webcam/camera access)."""
    # 1. Try PySide6 screen grab if running inside Qt application (100% reliable inside GUI)
    try:
        from PySide6.QtCore import QBuffer, QIODevice, Qt
        from PySide6.QtGui import QGuiApplication
        app = QGuiApplication.instance()
        if app:
            screen = app.primaryScreen()
            if screen:
                pixmap = screen.grabWindow(0)
                if not pixmap.isNull():
                    scaled = pixmap.scaled(1280, 720, Qt.KeepAspectRatio, Qt.SmoothTransformation)
                    qbuf = QBuffer()
                    qbuf.open(QIODevice.WriteOnly)
                    scaled.save(qbuf, "JPEG", 78)
                    return bytes(qbuf.data())
    except Exception:
        pass

    # 2. Try mss screen capture
    if HAS_SCREEN:
        try:
            with mss.mss() as sct:
                mon = sct.monitors[1] if len(sct.monitors) > 1 else sct.monitors[0]
                raw = sct.grab(mon)
                img = Image.frombytes("RGB", raw.size, raw.bgra, "raw", "BGRX")
                img.thumbnail((1280, 720))
                buf = io.BytesIO()
                img.save(buf, format="JPEG", quality=78)
                return buf.getvalue()
        except Exception:
            pass

    # 3. Try PIL ImageGrab fallback
    try:
        from PIL import ImageGrab
        img = ImageGrab.grab()
        img.thumbnail((1280, 720))
        buf = io.BytesIO()
        img.save(buf, format="JPEG", quality=78)
        return buf.getvalue()
    except Exception:
        pass

    return None

"""
A-3: Native Windows 11 UI Automation (UIA) Accessibility Tree Agent
===================================================================
Inspired by CursorTouch/Windows-MCP & Windows-Use.
Replaces brittle pixel/coordinate clicking with native Windows UI Automation:
  - Reads the live semantic Accessibility Tree of any Windows desktop app
    (WhatsApp Desktop, File Explorer, VS Code, Notepad, Settings, Excel, etc.)
  - Indexes every interactive control ([1] Button: Send, [2] Edit: Search...)
  - Clicks, types, expands, and verifies controls by index or accessible name
  - 100% immune to DPI scaling, multi-monitor offsets, or window resizing.

Supports `uiautomation` (fastest native COM wrapper) and `pywinauto` (UIA backend)
with automatic graceful fallback so DarkShadow never crashes if optional packages
are not yet installed.
"""

from __future__ import annotations

import time
from dataclasses import dataclass, field
from typing import Any

from error_tracker import log_error

IS_WINDOWS = platform.system().lower() == "windows"

_UIA_BACKEND = "none"
auto: Any = None
Desktop: Any = None

if IS_WINDOWS:
    try:
        import uiautomation as auto  # type: ignore
        _UIA_BACKEND = "uiautomation"
    except Exception:
        auto = None

    if _UIA_BACKEND == "none":
        try:
            import warnings
            with warnings.catch_warnings():
                warnings.filterwarnings("ignore", category=UserWarning, message=".*STA COM.*")
                from pywinauto import Desktop  # type: ignore
            _UIA_BACKEND = "pywinauto"
        except Exception:

INTERACTIVE_CONTROL_TYPES = {

@dataclass
class UIElementRef:
    index: int
    control_type: str
    name: str
    automation_id: str = ""
    value: str = ""
    rect: tuple[int, int, int, int] = (0, 0, 0, 0)
    handle: Any = field(default=None, repr=False)

    def to_compact_line(self) -> str:
        short_type = self.control_type.replace("Control", "")
        val_part = f' = "{self.value[:40]}"' if self.value else ""
        return f"[{self.index}] {short_type}: {self.name}{val_part}"


class WindowsUIAAgent:
    """
    Deterministic Windows 11 Accessibility Tree controller.
    Caches the most recent inspected element map so Gemini/LangGraph can say:
      1) inspect_window(window_title="WhatsApp")
      2) click_element(index=14) or type_into_element(index=8, text="Hello")
    """

    def __init__(self) -> None:
        self._last_window_title: str = ""
        self._last_elements: dict[int, UIElementRef] = {}
        self._last_snapshot_ts: float = 0.0

    def backend_info(self) -> dict[str, Any]:
        return {
            "is_windows": IS_WINDOWS,
            "backend": _UIA_BACKEND,
            "ready": IS_WINDOWS and _UIA_BACKEND != "none",
            "install_hint": "pip install uiautomation pywinauto" if _UIA_BACKEND == "none" else "",
        }

    def list_open_windows(self, limit: int = 25) -> dict[str, Any]:
        if not IS_WINDOWS:
            return {"ok": False, "message": "Windows UI Automation requires Windows 11."}
        if _UIA_BACKEND == "none":
            return {
                "ok": False,
                "message": "Install `uiautomation` or `pywinauto` first (run update_install_deps.bat).",
            }

        titles: list[dict[str, Any]] = []
        try:
            if _UIA_BACKEND == "uiautomation":
                with auto.UIAutomationInitializerInThread():
                    root = auto.GetRootControl()
                    for win in root.GetChildren():
                        name = (win.Name or "").strip()
                        if not name or win.IsOffscreen:
                            continue
                        rect = win.BoundingRectangle
                        if rect.width() < 80 or rect.height() < 60:
                            continue
                        titles.append({
                            "title": name,
                            "class_name": win.ClassName or "",
                            "pid": int(getattr(win, "ProcessId", 0) or 0),
                        })
                        if len(titles) >= limit:
                            break
            else:
                for win in Desktop(backend="uia").windows():
                    name = (win.window_text() or "").strip()
                    if not name:
                        continue
                    titles.append({
                        "title": name,
                        "class_name": win.class_name() or "",
                        "pid": int(win.process_id()),
                    })
                    if len(titles) >= limit:
                        break
            return {
                "ok": True,
                "backend": _UIA_BACKEND,
                "count": len(titles),
                "windows": titles,
                "message": f"Found {len(titles)} visible Windows apps.",
            }
        except Exception as exc:
            log_error("windows_uia_agent", f"list_open_windows failed: {exc}")
            return {"ok": False, "message": f"Could not enumerate windows: {exc}"}

    def _find_target_window_uia(self, window_title: str | None) -> Any:
        if not window_title:
            return auto.GetForegroundControl()
        target_l = window_title.lower().strip()
        root = auto.GetRootControl()
        for win in root.GetChildren():
            if target_l in (win.Name or "").lower():
                return win
        return auto.GetForegroundControl()

    def _find_target_window_pywinauto(self, window_title: str | None) -> Any:
        desktop = Desktop(backend="uia")
        if not window_title:
            return desktop.top_window()
        target_l = window_title.lower().strip()
        for win in desktop.windows():
            if target_l in (win.window_text() or "").lower():
                return win
        return desktop.top_window()

    def focus_window(self, window_title: str) -> dict[str, Any]:
        if not IS_WINDOWS or _UIA_BACKEND == "none":
            return {"ok": False, "message": "Windows UIA backend unavailable."}
        try:
            if _UIA_BACKEND == "uiautomation":
                with auto.UIAutomationInitializerInThread():
                    win = self._find_target_window_uia(window_title)
                    if not win:
                        return {"ok": False, "message": f"Window '{window_title}' not found."}
                    win.SetActive()
                    try:
                        win.SetTopmost(False)
                    except Exception:
                        pass
                    return {
                        "ok": True,
                        "window": win.Name,
                        "message": f"Focused window '{win.Name}'.",
                    }
            else:
                win = self._find_target_window_pywinauto(window_title)
                win.set_focus()
                return {
                    "ok": True,
                    "window": win.window_text(),
                    "message": f"Focused window '{win.window_text()}'.",
                }
        except Exception as exc:
            log_error("windows_uia_agent", f"focus_window failed: {exc}")
            return {"ok": False, "message": f"Focus window failed: {exc}"}

    def inspect_window(
        self,
        window_title: str | None = None,
        max_elements: int = 60,
        max_depth: int = 6,
    ) -> dict[str, Any]:
        """
        Walks the Windows UI Automation Accessibility Tree and returns a compact,
        numbered list of interactive elements ready for LLM tool calling.
        """
        if not IS_WINDOWS:
            return {"ok": False, "message": "Windows UI Automation requires Windows 11."}
        if _UIA_BACKEND == "none":
            return {
                "ok": False,
                "message": "Install `uiautomation` or `pywinauto` first (run update_install_deps.bat).",
            }

        elements: dict[int, UIElementRef] = {}
        idx = 1

        try:
            if _UIA_BACKEND == "uiautomation":
                with auto.UIAutomationInitializerInThread():
                    win = self._find_target_window_uia(window_title)
                    if not win:
                        return {"ok": False, "message": "No active window found."}
                    win_name = (win.Name or "Active Window").strip()

                    queue: list[tuple[Any, int]] = [(win, 0)]
                    while queue and idx <= max_elements:
                        ctrl, depth = queue.pop(0)
                        if depth > max_depth:
                            continue
                        ctype = getattr(ctrl, "ControlTypeName", "") or ""
                        cname = (getattr(ctrl, "Name", "") or "").strip()
                        auto_id = (getattr(ctrl, "AutomationId", "") or "").strip()

                        if depth > 0 and ctype in INTERACTIVE_CONTROL_TYPES and (cname or auto_id):
                            if not getattr(ctrl, "IsOffscreen", False):
                                val = ""
                                try:
                                    vp = ctrl.GetValuePattern()
                                    if vp:
                                        val = (vp.Value or "").strip()
                                except Exception:
                                    pass
                                rect_obj = ctrl.BoundingRectangle
                                rect = (
                                    int(rect_obj.left),
                                    int(rect_obj.top),
                                    int(rect_obj.right),
                                    int(rect_obj.bottom),
                                )
                                elements[idx] = UIElementRef(
                                    index=idx,
                                    control_type=ctype,
                                    name=cname or auto_id,
                                    automation_id=auto_id,
                                    value=val,
                                    rect=rect,
                                    handle=None,  # Re-resolved safely per thread on click
                                )
                                idx += 1

                        try:
                            for child in ctrl.GetChildren():
                                queue.append((child, depth + 1))
                        except Exception:
                            continue
            else:
                win = self._find_target_window_pywinauto(window_title)
                win_name = (win.window_text() or "Active Window").strip()
                for desc in win.descendants():
                    if idx > max_elements:
                        break
                    ctype = desc.element_info.control_type or ""
                    cname = (desc.window_text() or desc.element_info.name or "").strip()
                    auto_id = (desc.element_info.automation_id or "").strip()
                    if ctype in INTERACTIVE_CONTROL_TYPES and (cname or auto_id):
                        r = desc.rectangle()
                        elements[idx] = UIElementRef(
                            index=idx,
                            control_type=ctype,
                            name=cname or auto_id,
                            automation_id=auto_id,
                            rect=(int(r.left), int(r.top), int(r.right), int(r.bottom)),
                            handle=desc,
                        )
                        idx += 1

            self._last_window_title = win_name
            self._last_elements = elements
            self._last_snapshot_ts = time.time()

            tree_lines = [el.to_compact_line() for el in elements.values()]
            return {
                "ok": True,
                "backend": _UIA_BACKEND,
                "window": win_name,
                "element_count": len(elements),
                "tree": "\n".join(tree_lines),
                "message": f"Inspected '{win_name}': {len(elements)} interactive UI elements indexed.",
            }
        except Exception as exc:
            log_error("windows_uia_agent", f"inspect_window failed: {exc}")
            return {"ok": False, "message": f"UI tree inspection failed: {exc}"}

    def _resolve_target(self, index: int | None, name: str | None) -> UIElementRef | None:
        if index is not None and index in self._last_elements:
            return self._last_elements[index]
        if name:
            needle = name.lower().strip()
            # Exact match first
            for el in self._last_elements.values():
                if el.name.lower() == needle:
                    return el
            # Substring match fallback
            for el in self._last_elements.values():
                if needle in el.name.lower() or needle in el.automation_id.lower():
                    return el
        return None

    def click_element(
        self,
        index: int | None = None,
        name: str | None = None,
        window_title: str | None = None,
    ) -> dict[str, Any]:
        """Clicks a Windows UI element by indexed ref or accessible name."""
        if not self._last_elements or (
            window_title and window_title.lower() not in self._last_window_title.lower()
        ):
            self.inspect_window(window_title=window_title)

        target = self._resolve_target(index, name)
        if not target:
            return {
                "ok": False,
                "message": f"Element (index={index}, name={name}) not found in '{self._last_window_title}'. Run inspect first.",
            }

        try:
            if _UIA_BACKEND == "uiautomation":
                with auto.UIAutomationInitializerInThread():
                    # Click center of verified UIA bounding box (immune to DPI offsets via UIA)
                    cx = (target.rect[0] + target.rect[2]) // 2
                    cy = (target.rect[1] + target.rect[3]) // 2
                    if cx <= 0 and cy <= 0:
                        return {"ok": False, "message": f"Element '{target.name}' has invalid bounds."}
                    auto.Click(cx, cy)
            elif _UIA_BACKEND == "pywinauto" and target.handle is not None:
                target.handle.click_input()
            else:
                return {"ok": False, "message": "No active UIA backend available."}

            return {
                "ok": True,
                "window": self._last_window_title,
                "clicked": target.to_compact_line(),
                "message": f"Clicked {target.to_compact_line()} in '{self._last_window_title}'.",
            }
        except Exception as exc:
            log_error("windows_uia_agent", f"click_element failed: {exc}")
            return {"ok": False, "message": f"Click failed: {exc}"}

    def type_into_element(
        self,
        text: str,
        index: int | None = None,
        name: str | None = None,
        press_enter: bool = False,
        window_title: str | None = None,
    ) -> dict[str, Any]:
        """Focuses a Windows input/edit control and types text cleanly."""
        click_res = self.click_element(index=index, name=name, window_title=window_title)
        if not click_res.get("ok"):
            return click_res

        try:
            time.sleep(0.15)
            if _UIA_BACKEND == "uiautomation":
                with auto.UIAutomationInitializerInThread():
                    auto.SendKeys(text, interval=0.015)
                    if press_enter:
                        auto.SendKeys("{Enter}")
            elif _UIA_BACKEND == "pywinauto":
                target = self._resolve_target(index, name)
                if target and target.handle is not None:
                    target.handle.type_keys(text + ("{ENTER}" if press_enter else ""), with_spaces=True)

            return {
                "ok": True,
                "window": self._last_window_title,
                "typed_preview": text[:80],
                "pressed_enter": press_enter,
                "message": f"Typed into {click_res.get('clicked')} in '{self._last_window_title}'.",
            }
        except Exception as exc:
            log_error("windows_uia_agent", f"type_into_element failed: {exc}")
            return {"ok": False, "message": f"Typing failed: {exc}"}

"""
D-1 & A-1/A-3: Universal Model Context Protocol (MCP) Client Hub
================================================================
Turns DarkShadow into a real MCP Client using the official `mcp` Python SDK.
Reads `mcp_servers.json` and dynamically connects to external MCP servers:
  - `@playwright/mcp` (Official Microsoft Accessibility Tree Browser MCP)
  - `windows-mcp` (CursorTouch native Windows 11 MCP)
  - Any community MCP server (GitHub, Gmail, Notion, Filesystem, SQLite, etc.)

All outbound MCP tool calls pass through DarkShadow's centralized Safety Shield
so external servers can NEVER execute payments, OTP entry, or destructive shell commands.
"""

from __future__ import annotations

import asyncio
import concurrent.futures
import json
import os
import re
from pathlib import Path
from typing import Any

from error_tracker import log_error

DEFAULT_CONFIG_PATH = Path(__file__).resolve().parent / "mcp_servers.json"

_BLOCKED_ARG_RE = re.compile(
    r"\b(pay now|place order|confirm purchase|cvv|upi pin|format\s+c:|rm\s+-rf|del\s+/s\s+/q)\b",
    re.IGNORECASE,
)


class MCPClientHub:
    """
    Lightweight on-demand MCP client manager.
    Connects via stdio transport per server (avoids dangling zombie Node/uvx subprocesses
    when DarkShadow exits) and caches discovered tool metadata.
    """

    def __init__(self, config_path: Path | None = None) -> None:
        self.config_path = config_path or DEFAULT_CONFIG_PATH
        self._tool_cache: dict[str, list[dict[str, Any]]] = {}

    def load_config(self) -> dict[str, dict[str, Any]]:
        if not self.config_path.exists():
            return {}
        try:
            raw = json.loads(self.config_path.read_text(encoding="utf-8"))
            servers = raw.get("mcpServers", {})
            if isinstance(servers, dict):
                return servers
        except Exception as exc:
            log_error("mcp_client_hub", f"Failed to parse {self.config_path}: {exc}")
        return {}

    def list_configured_servers(self) -> dict[str, Any]:
        servers = self.load_config()
        summary = [
            {
                "name": name,
                "enabled": bool(cfg.get("enabled", False)),
                "command": cfg.get("command", ""),
                "description": cfg.get("description", ""),
            }
            for name, cfg in servers.items()
        ]
        return {
            "ok": True,
            "config_path": str(self.config_path),
            "servers": summary,
            "message": f"{len(summary)} MCP servers configured ({sum(1 for s in summary if s['enabled'])} enabled).",
        }

    def _is_sdk_installed(self) -> bool:
        try:
            import mcp  # noqa: F401
            return True
        except Exception:
            return False

    async def discover_tools_async(self, server_name: str) -> dict[str, Any]:
        if not self._is_sdk_installed():
            return {
                "ok": False,
                "message": "Official `mcp` SDK not installed. Run `pip install mcp` (or update_install_deps.bat).",
            }

        servers = self.load_config()
        cfg = servers.get(server_name)
        if not cfg:
            return {"ok": False, "message": f"MCP server '{server_name}' not found in mcp_servers.json."}
        if not cfg.get("enabled", False):
            return {
                "ok": False,
                "message": f"MCP server '{server_name}' is disabled in mcp_servers.json. Set enabled: true to activate.",
            }

        from mcp import ClientSession, StdioServerParameters  # type: ignore
        from mcp.client.stdio import stdio_client  # type: ignore

        params = StdioServerParameters(
            command=str(cfg["command"]),
            args=[str(a) for a in cfg.get("args", [])],
            env={**os.environ, **{str(k): str(v) for k, v in (cfg.get("env") or {}).items()}},
        )

        try:
            async with stdio_client(params) as (read_stream, write_stream):
                async with ClientSession(read_stream, write_stream) as session:
                    await session.initialize()
                    result = await session.list_tools()
                    tools = [
                        {
                            "name": t.name,
                            "description": (t.description or "")[:180],
                            "inputSchema": getattr(t, "inputSchema", {}),
                        }
                        for t in (result.tools or [])
                    ]
                    self._tool_cache[server_name] = tools
                    return {
                        "ok": True,
                        "server": server_name,
                        "tool_count": len(tools),
                        "tools": tools,
                        "message": f"Discovered {len(tools)} tools on MCP server '{server_name}'.",
                    }
        except Exception as exc:
            log_error("mcp_client_hub", f"discover_tools({server_name}) failed: {exc}")
            return {"ok": False, "server": server_name, "message": f"MCP discovery failed: {exc}"}

    async def call_tool_async(
        self,
        server_name: str,
        tool_name: str,
        arguments: dict[str, Any] | None = None,
    ) -> dict[str, Any]:
        if not self._is_sdk_installed():
            return {
                "ok": False,
                "message": "Official `mcp` SDK not installed. Run `pip install mcp`.",
            }

        args_payload = arguments or {}
        serialized_args = json.dumps(args_payload, ensure_ascii=False)
        if _BLOCKED_ARG_RE.search(f"{tool_name} {serialized_args}"):
            return {
                "ok": False,
                "blocked": True,
                "message": "Captain, Safety Shield ne risky/payment MCP tool call block kar diya.",
            }

        servers = self.load_config()
        cfg = servers.get(server_name)
        if not cfg:
            return {"ok": False, "message": f"MCP server '{server_name}' not found in mcp_servers.json."}
        if not cfg.get("enabled", False):
            return {
                "ok": False,
                "message": f"MCP server '{server_name}' is disabled in mcp_servers.json.",
            }

        from mcp import ClientSession, StdioServerParameters  # type: ignore
        from mcp.client.stdio import stdio_client  # type: ignore

        params = StdioServerParameters(
            command=str(cfg["command"]),
            args=[str(a) for a in cfg.get("args", [])],
            env={**os.environ, **{str(k): str(v) for k, v in (cfg.get("env") or {}).items()}},
        )

        try:
            async with stdio_client(params) as (read_stream, write_stream):
                async with ClientSession(read_stream, write_stream) as session:
                    await session.initialize()
                    res = await session.call_tool(tool_name, arguments=args_payload)
                    text_parts: list[str] = []
                    for item in getattr(res, "content", []) or []:
                        text = getattr(item, "text", None)
                        if text:
                            text_parts.append(str(text))
                    output = "\n".join(text_parts).strip()
                    is_err = bool(getattr(res, "isError", False))
                    return {
                        "ok": not is_err,
                        "server": server_name,
                        "tool": tool_name,
                        "output": output[:3000],
                        "message": output[:400] or f"Executed {server_name}.{tool_name}.",
                    }
        except Exception as exc:
            log_error("mcp_client_hub", f"call_tool({server_name}.{tool_name}) failed: {exc}")
            return {
                "ok": False,
                "server": server_name,
                "tool": tool_name,
                "message": f"MCP call error: {exc}",
            }

    def _run_sync(self, coro: Any, timeout: int = 90) -> dict[str, Any]:
        try:
            loop = asyncio.get_running_loop()
        except RuntimeError:
            loop = None
        if loop and loop.is_running():
            with concurrent.futures.ThreadPoolExecutor(max_workers=1) as pool:
                return pool.submit(lambda: asyncio.run(coro)).result(timeout=timeout)
        return asyncio.run(coro)

    def discover_tools(self, server_name: str) -> dict[str, Any]:
        return self._run_sync(self.discover_tools_async(server_name))

    def call_tool(
        self,
        server_name: str,
        tool_name: str,
        arguments: dict[str, Any] | None = None,
    ) -> dict[str, Any]:
        return self._run_sync(self.call_tool_async(server_name, tool_name, arguments))

"""
A-1 & A-2: Autonomous Web Agent (`browser-use` + Playwright Accessibility Tree CDP)
===================================================================================
Combines the two highest-accuracy browser automation paradigms in one clean module:
  1. Fast Accessibility-Tree Mode (Playwright CDP on Microsoft Edge port 9222):
     Extracts compact semantic accessibility snapshots (zero heavy screenshot bloat)
     and executes deterministic indexed actions.
  2. Full Autonomous Multi-Step Web Mode (`browser-use` Agent):
     Attaches directly to Captain's logged-in Microsoft Edge CDP session and runs
     autonomous multi-step web research, comparisons, form filling, YouTube/Spotify
     navigation, and structured extraction using Gemini.

Enforces DarkShadow's unbypassable Payment/Checkout Exclusion Shield before every task.
"""

from __future__ import annotations

import asyncio
import os
import re
from typing import Any

from browser_agent import CDP_URL, EdgeBrowserAgent
from error_tracker import log_error

EDGE_AGENT = EdgeBrowserAgent(cdp_url=CDP_URL)

# Unbypassable payment / checkout / OTP safety filter
_PAYMENT_BLOCK_RE = re.compile(
    r"\b(pay now|complete payment|place order|confirm purchase|enter otp|enter cvv|"
    r"upi pin|netbanking login|credit card number|debit card number|buy now and pay|"
    r"transfer money|send money|bank password)\b",
    re.IGNORECASE,

def _is_blocked_web_goal(goal: str) -> bool:
    return bool(_PAYMENT_BLOCK_RE.search(goal or ""))


class AutonomousWebAgent:
    """
    Unified high-level web automation engine for DarkShadow:
      - `get_accessibility_tree()`: ultra-lightweight Playwright A11y tree snapshot
      - `run_autonomous_goal(goal, max_steps)`: runs `browser-use` over Captain's Edge CDP
        with automatic graceful fallback to Playwright DOM/Accessibility agent if `browser-use`
        is not installed yet.
    """

    def __init__(self, cdp_url: str = CDP_URL) -> None:
        self.cdp_url = cdp_url

    def check_capabilities(self) -> dict[str, Any]:
        has_browser_use = False
        has_playwright = False
        try:
            import browser_use  # noqa: F401
            has_browser_use = True
        except Exception:
            pass
        try:
            import playwright  # noqa: F401
            has_playwright = True
        except Exception:
            pass
        return {
            "cdp_url": self.cdp_url,
            "browser_use_installed": has_browser_use,
            "playwright_installed": has_playwright,
            "install_hint": (
                "pip install browser-use langchain-google-genai playwright"
                if not has_browser_use
                else ""
            ),
        }

    def get_accessibility_tree(self, max_nodes: int = 80) -> dict[str, Any]:
        """
        A-1: Reads the current Microsoft Edge tab via Playwright's Accessibility Tree
        (same principle as @microsoft/playwright-mcp — fast, lightweight, semantic).
        """
        try:
            from playwright.sync_api import sync_playwright  # type: ignore
        except Exception:
            return EDGE_AGENT.list_interactive_elements()

        lines: list[str] = []
        try:
            with sync_playwright() as p:
                browser = p.chromium.connect_over_cdp(self.cdp_url, timeout=4000)
                if not browser.contexts or not browser.contexts[0].pages:
                    return {"ok": False, "message": "No open Microsoft Edge tabs found on CDP 9222."}
                page = browser.contexts[0].pages[-1]
                title = page.title()
                url = page.url

                # Extract compact ARIA/Accessibility tree via modern aria_snapshot when available,
                # falling back to accessibility.snapshot() or DOM interactive map.
                aria_text = ""
                try:
                    aria_text = page.locator("body").aria_snapshot(timeout=3000)
                except Exception:
                    try:
                        snap = page.accessibility.snapshot(interesting_only=True)  # type: ignore[attr-defined]
                        if snap:
                            self._flatten_a11y(snap, lines, max_nodes)
                            aria_text = "\n".join(lines)
                    except Exception:
                        pass

                if not aria_text:
                    return EDGE_AGENT.list_interactive_elements()

                trimmed_lines = aria_text.splitlines()[:max_nodes]
                return {
                    "ok": True,
                    "mode": "playwright_accessibility_tree",
                    "title": title,
                    "url": url,
                    "node_count": len(trimmed_lines),
                    "tree": "\n".join(trimmed_lines),
                    "message": f"Read Accessibility Tree for '{title}' ({len(trimmed_lines)} nodes).",
                }
        except Exception as exc:
            log_error("autonomous_web_agent", f"get_accessibility_tree failed: {exc}")
            # Fall back to standard Edge DOM snapshot
            return EDGE_AGENT.list_interactive_elements()

    def _flatten_a11y(self, node: dict[str, Any], out: list[str], limit: int, depth: int = 0) -> None:
        if len(out) >= limit:
            return
        role = node.get("role", "")
        name = (node.get("name") or "").strip()
        if role and name:
            indent = "  " * min(depth, 4)
            out.append(f"{indent}- {role}: {name[:90]}")
        for child in node.get("children", []) or []:
            self._flatten_a11y(child, out, limit, depth + 1)

    async def run_autonomous_goal_async(
        self,
        goal: str,
        max_steps: int = 10,
        start_url: str | None = None,
    ) -> dict[str, Any]:
        """
        A-2: Runs a full autonomous multi-step browser goal via `browser-use`
        attached directly to Captain's Microsoft Edge CDP session (port 9222).
        """
        if _is_blocked_web_goal(goal):
            return {
                "ok": False,
                "blocked": True,
                "message": "Captain, Payment/Checkout/OTP Exclusion Shield active hai — autonomous checkout allowed nahi hai.",
            }

        api_key = os.getenv("GEMINI_API_KEY", "").strip()
        if not api_key:
            return {"ok": False, "message": "GEMINI_API_KEY missing in .env."}

        try:
            from browser_use import Agent, Browser  # type: ignore
        except Exception:
            # Graceful guidance + fallback using deterministic Edge CDP search/navigation
            if start_url:
                nav = EDGE_AGENT.navigate_active_tab(start_url)
                return {
                    "ok": bool(nav.get("ok")),
                    "mode": "edge_cdp_fallback",
                    "message": (
                        f"`browser-use` package abhi install nahi hai (run update_install_deps.bat). "
                        f"Filhaal Edge mein {start_url} open kar diya hai."
                    ),
                }
            return {
                "ok": False,
                "mode": "missing_browser_use",
                "message": (
                    "`browser-use` install nahi hai. `update_install_deps.bat` chalayein "
                    "(installs browser-use + playwright) taaki autonomous multi-step web tasks chal sakein."
                ),
            }

        # Pick best available Gemini LLM adapter supported by the installed browser-use version
        llm: Any = None
        model_name = os.getenv("DARKSHADOW_BROWSER_MODEL", "gemini-4-hydrogen")
        try:
            from browser_use.llm import ChatGoogle  # type: ignore
            llm = ChatGoogle(model=model_name, api_key=api_key)
        except Exception:
            try:
                from langchain_google_genai import ChatGoogleGenerativeAI  # type: ignore
                llm = ChatGoogleGenerativeAI(model=model_name, google_api_key=api_key)
            except Exception as exc:
                log_error("autonomous_web_agent", f"LLM init failed: {exc}")
                return {
                    "ok": False,
                    "message": "Install `langchain-google-genai` or latest `browser-use` for Gemini support.",
                }

        browser: Any = None
        try:
            # Attach directly to Captain's existing Edge CDP session (logged-in cookies stay intact!)
            try:
                browser = Browser(cdp_url=self.cdp_url)
            except TypeError:
                from browser_use import BrowserConfig  # type: ignore
                browser = Browser(config=BrowserConfig(cdp_url=self.cdp_url))

            full_task = (
                f"{goal}\n"
                "STRICT SAFETY RULE: NEVER complete payments, checkout, enter passwords/OTP/CVV, "
                "or submit financial transfers. Stop right before any payment/checkout step."
            )
            if start_url:
                full_task = f"First open {start_url}. Then: {full_task}"

            agent = Agent(task=full_task, llm=llm, browser=browser)
            history = await agent.run(max_steps=max(2, min(max_steps, 20)))

            final_text = ""
            if hasattr(history, "final_result") and callable(history.final_result):
                final_text = str(history.final_result() or "").strip()
            if not final_text:
                final_text = str(history)[:800]

            return {
                "ok": True,
                "mode": "browser_use_cdp",
                "goal": goal,
                "result": final_text,
                "message": final_text or f"Autonomous browser goal completed: {goal}",
            }
        except Exception as exc:
            log_error("autonomous_web_agent", f"browser-use execution error: {exc}")
            return {
                "ok": False,
                "mode": "browser_use_error",
                "message": (
                    f"Autonomous browser agent error: {exc}. "
                    "Make sure Launch_Edge_Debug.bat is running on port 9222."
                ),
            }
        finally:
            # Do NOT kill Captain's live Edge browser — only release CDP handle if supported
            if browser is not None:
                try:
                    close_fn = getattr(browser, "stop", None) or getattr(browser, "close", None)
                    if callable(close_fn):
                        maybe_coro = close_fn()
                        if asyncio.iscoroutine(maybe_coro):
                            await maybe_coro
                except Exception:
                    pass

    def run_autonomous_goal(
        self,
        goal: str,
        max_steps: int = 10,
        start_url: str | None = None,
    ) -> dict[str, Any]:
        """Synchronous wrapper safe to invoke from worker threads."""
        try:
            loop = asyncio.get_running_loop()
        except RuntimeError:
            loop = None

        if loop and loop.is_running():
            # Execute in isolated thread event loop to prevent nested-loop errors
            import concurrent.futures
            with concurrent.futures.ThreadPoolExecutor(max_workers=1) as pool:
                return pool.submit(
                    lambda: asyncio.run(
                        self.run_autonomous_goal_async(goal, max_steps=max_steps, start_url=start_url)
                    )
                ).result(timeout=180)
        return asyncio.run(
            self.run_autonomous_goal_async(goal, max_steps=max_steps, start_url=start_url)
        )

"""
C-All: Super-Human Long-Term Memory (Mem0 + Letta-Style Virtual Context Memory)
===============================================================================
Upgrades DarkShadow's memory into a 3-tier OS-inspired cognitive hierarchy:
  1. Core RAM Blocks (Letta / MemGPT pattern):
     Always-loaded editable JSON memory blocks (`captain_profile`, `preferences`,
     `active_projects`, `persona_notes`) injected straight into the system prompt.
     DarkShadow can autonomously call `core_memory_append` or `core_memory_replace`!
  2. Living Fact Graph (`mem0ai` / Mem0):
     Automatically extracts, deduplicates, and updates long-term user preferences
     and habits from every conversation turn using Gemini + local vector storage
     inside `.darkshadow/mem0_qdrant` (100% private & local).
  3. Contradiction-Free SQLite Archival + Reminders (`memory_engine.py`):
     Preserves 100% backwards compatibility with Captain's existing `.darkshadow/memory.db`!

Zero-crash guarantee: if `mem0ai` is not installed yet, it transparently runs
on Core RAM Blocks + our contradiction-free SQLite SemanticMemory!
"""

from __future__ import annotations

import json
import os
from datetime import datetime
from pathlib import Path
from typing import Any

from error_tracker import log_error
from memory_engine import SemanticMemoryEngine

DEFAULT_STORAGE_DIR = Path.home() / ".darkshadow"
CORE_BLOCKS_FILE = "core_memory_blocks.json"

DEFAULT_CORE_BLOCKS: dict[str, str] = {
    "captain_profile": "Name: Captain Kuldeep. OS: Windows 11. Preferred AI name: DarkShadow (Hriday).",
    "preferences": "Prefers concise, direct Hindi/Hinglish/Bhojpuri/English replies. Uses Microsoft Edge.",
    "active_projects": "Building DarkShadow — Jarvis-level Windows 11 AI co-pilot with voice, browser & OS automation.",
    "safety_rules": "Payment/Checkout/OTP strictly forbidden. Physical webcam permanently disabled.",
}


class LivingMemoryV3:
    """
    Unified 3-tier Memory Manager:
      - Tier 1 (RAM): Letta-style self-editable Core Memory Blocks
      - Tier 2 (Living Facts): Mem0 auto-extraction & conflict resolution (optional `mem0ai`)
      - Tier 3 (Disk/Archival): Existing SQLite `.darkshadow/memory.db` via SemanticMemoryEngine
    """

    def __init__(
        self,
        semantic_memory: SemanticMemoryEngine | None = None,
        storage_dir: Path | None = None,
        user_id: str = "captain_kuldeep",
    ) -> None:
        self.storage_dir = storage_dir or DEFAULT_STORAGE_DIR
        self.storage_dir.mkdir(parents=True, exist_ok=True)
        self.user_id = user_id
        self.semantic = semantic_memory or SemanticMemoryEngine(self.storage_dir / "memory.db")
        self._blocks_path = self.storage_dir / CORE_BLOCKS_FILE
        self._core_blocks: dict[str, str] = self._load_core_blocks()
        self._mem0: Any = None
        self._mem0_status: str = "not_initialized"
        self._init_mem0_lazy()

    # ------------------------------------------------------------------
    # Tier 1: Letta-style Core RAM Memory Blocks
    # ------------------------------------------------------------------
    def _load_core_blocks(self) -> dict[str, str]:
        if self._blocks_path.exists():
            try:
                data = json.loads(self._blocks_path.read_text(encoding="utf-8"))
                if isinstance(data, dict):
                    merged = dict(DEFAULT_CORE_BLOCKS)
                    for k, v in data.items():
                        merged[str(k)] = str(v)
                    return merged
            except Exception as exc:
                log_error("living_memory_v3", f"Core blocks load failed: {exc}")
        self._save_core_blocks(DEFAULT_CORE_BLOCKS)
        return dict(DEFAULT_CORE_BLOCKS)

    def _save_core_blocks(self, blocks: dict[str, str]) -> None:
        try:
            self._blocks_path.write_text(
                json.dumps(blocks, indent=2, ensure_ascii=False),
                encoding="utf-8",
            )
        except Exception as exc:
            log_error("living_memory_v3", f"Core blocks save failed: {exc}")

    def get_core_blocks(self) -> dict[str, str]:
        return dict(self._core_blocks)

    def core_memory_append(self, block: str, content: str) -> dict[str, Any]:
        key = block.strip().lower()
        clean = content.strip()
        if not key or not clean:
            return {"ok": False, "message": "Block name and content are required."}
        existing = self._core_blocks.get(key, "").strip()
        if clean.lower() in existing.lower():
            return {"ok": True, "block": key, "message": f"Already stored in core block '{key}'."}
        updated = f"{existing} {clean}".strip() if existing else clean
        # Keep core RAM blocks crisp (< 1200 chars per block)
        if len(updated) > 1200:
            updated = updated[-1200:]
        self._core_blocks[key] = updated
        self._save_core_blocks(self._core_blocks)
        return {
            "ok": True,
            "block": key,
            "value": updated,
            "message": f"Appended to core RAM block '{key}'.",
        }

    def core_memory_replace(self, block: str, old_text: str, new_text: str) -> dict[str, Any]:
        key = block.strip().lower()
        existing = self._core_blocks.get(key, "")
        if not existing:
            self._core_blocks[key] = new_text.strip()
            self._save_core_blocks(self._core_blocks)
            return {"ok": True, "block": key, "message": f"Created core RAM block '{key}'."}

        if old_text and old_text in existing:
            updated = existing.replace(old_text, new_text.strip(), 1)
        else:
            updated = new_text.strip()

        self._core_blocks[key] = updated[:1200]
        self._save_core_blocks(self._core_blocks)
        return {
            "ok": True,
            "block": key,
            "value": self._core_blocks[key],
            "message": f"Updated core RAM block '{key}'.",
        }

    # ------------------------------------------------------------------
    # Tier 2: Mem0 Automatic Living Memory (with Gemini backend)
    # ------------------------------------------------------------------
    def _init_mem0_lazy(self) -> None:
        api_key = os.getenv("GEMINI_API_KEY", "").strip()
        if not api_key:
            self._mem0_status = "missing_gemini_api_key"
            return
        try:
            from mem0 import Memory  # type: ignore
        except Exception:
            self._mem0_status = "mem0_not_installed_using_sqlite_fallback"
            return

        qdrant_path = str(self.storage_dir / "mem0_qdrant")
        config = {
            "llm": {
                "provider": "gemini",
                "config": {
                    "model": os.getenv("DARKSHADOW_MEMORY_MODEL", "gemini-3.1-flash-lite"),
                    "api_key": api_key,
                    "temperature": 0.1,
                },
            },
            "embedder": {
                "provider": "gemini",
                "config": {
                    "model": "gemini-embedding-2",
                    "api_key": api_key,
                },
            },
            "vector_store": {
                "provider": "qdrant",
                "config": {
                    "collection_name": "darkshadow_captain_memory",
                    "path": qdrant_path,
                    "embedding_model_dims": 768,
                },
            },
        }
        try:
            self._mem0 = Memory.from_config(config)
            self._mem0_status = "mem0_active"
        except Exception as exc:
            log_error("living_memory_v3", f"Mem0 init fallback to SQLite: {exc}")
            self._mem0 = None
            self._mem0_status = f"sqlite_fallback ({type(exc).__name__})"

    # ------------------------------------------------------------------
    # Unified Public API
    # ------------------------------------------------------------------
    def status(self) -> dict[str, Any]:
        return {
            "ok": True,
            "mem0_status": self._mem0_status,
            "core_blocks": list(self._core_blocks.keys()),
            "storage_dir": str(self.storage_dir),
        }

    def remember(self, text: str, category: str = "preference") -> dict[str, Any]:
        """Stores a fact in BOTH Contradiction-Free SQLite AND Mem0 (when active)."""
        clean = text.strip()
        if not clean:
            return {"ok": False, "message": "Empty memory text."}

        # Always persist into existing SQLite SemanticMemoryEngine (keeps .darkshadow/memory.db synced!)
        sql_ok, sql_msg = self.semantic.save_memory(category, clean)

        mem0_saved = False
        if self._mem0 is not None:
            try:
                self._mem0.add(clean, user_id=self.user_id, metadata={"category": category})
                mem0_saved = True
            except Exception as exc:
                log_error("living_memory_v3", f"Mem0 add warning: {exc}")

        return {
            "ok": bool(sql_ok or mem0_saved),
            "mem0_synced": mem0_saved,
            "message": sql_msg,
        }

    def recall(self, query: str, limit: int = 5) -> dict[str, Any]:
        """Hybrid recall merging Mem0 living facts + SQLite semantic memories."""
        results: list[str] = []
        seen: set[str] = set()

        if self._mem0 is not None:
            try:
                res = self._mem0.search(query, user_id=self.user_id, limit=limit)
                items = res.get("results", []) if isinstance(res, dict) else (res or [])
                for item in items:
                    text = (item.get("memory") or item.get("text") or "").strip() if isinstance(item, dict) else str(item)
                    key = text.lower()
                    if text and key not in seen:
                        seen.add(key)
                        results.append(text)
            except Exception as exc:
                log_error("living_memory_v3", f"Mem0 search warning: {exc}")

        # Merge SQLite SemanticMemoryEngine recall
        try:
            sql_hits = self.semantic.recall_relevant(query, limit=limit)
            for hit in sql_hits:
                text = str(hit).strip()
                key = text.lower()
                if text and key not in seen:
                    seen.add(key)
                    results.append(text)
        except Exception as exc:
            log_error("living_memory_v3", f"SQLite recall warning: {exc}")

        return {
            "ok": True,
            "count": len(results),
            "memories": results[:limit],
            "message": "; ".join(results[:limit]) if results else "No matching memories found.",
        }

    def ingest_conversation_turn(self, user_text: str, assistant_text: str) -> None:
        """
        Automatically feeds completed turns into Mem0 so DarkShadow passively learns
        Captain's habits and updates outdated facts without explicit 'remember this' commands.
        """
        if self._mem0 is None:
            return
        if not user_text.strip() or len(user_text.strip()) < 6:
            return
        try:
            self._mem0.add(
                [
                    {"role": "user", "content": user_text.strip()},
                    {"role": "assistant", "content": assistant_text.strip()[:500]},
                ],
                user_id=self.user_id,
                metadata={"timestamp": datetime.utcnow().isoformat()},
            )
        except Exception as exc:
            log_error("living_memory_v3", f"Auto-ingest turn skipped: {exc}")

    def build_system_context_block(self, current_query: str = "") -> str:
        """Renders Letta-style Core RAM blocks + top relevant archival facts for system instructions."""
        sections = ["=== CORE MEMORY (RAM) ==="]
        for k, v in self._core_blocks.items():
            sections.append(f"[{k}]: {v}")

        if current_query.strip():
            recalled = self.recall(current_query, limit=4).get("memories", [])
            if recalled:
                sections.append("=== RELEVANT LONG-TERM MEMORIES ===")
                for item in recalled:
                    sections.append(f"- {item}")

        return "\n".join(sections)

"""
Upgrade #4: Contradiction-Free Living Semantic Memory, Reminders & Episodic Archive.
- Backed by SQLite + gemini-embedding-2 vectors + keyword hybrid recall.
- Automatically detects & updates conflicting/duplicate preferences instead of stacking contradictions.
- Preserves existing .darkshadow/memory.db tables cleanly (100% drop-in compatible).
- Uses active gemini-3.1-flash-lite for fast episodic session summaries.

from __future__ import annotations

import os
import re
import sqlite3
from dataclasses import dataclass
from datetime import datetime, timezone
from pathlib import Path
from typing import Any

import numpy as np

SENSITIVE_LABELS = re.compile(
    r"\b(?:api\s*key|password|passcode|otp|cvv|credit\s*card|debit\s*card|"
    r"bank\s*account|upi\s*pin|private\s*key|secret\s*key|access\s*token)\b",
    re.IGNORECASE,
)
SECRET_SHAPES = (
    re.compile(r"\bAIza[0-9A-Za-z_-]{20,}\b"),
    re.compile(r"\bsk-[0-9A-Za-z_-]{16,}\b", re.IGNORECASE),
    re.compile(r"-----BEGIN [A-Z ]*PRIVATE KEY-----"),
    re.compile(r"\b(?:\d[ -]?){13,19}\b"),
)


def contains_sensitive_data(text: str) -> bool:
    cleaned = str(text).strip()
    if SENSITIVE_LABELS.search(cleaned):
        return True
    return any(p.search(cleaned) for p in SECRET_SHAPES)


@dataclass(frozen=True)
class ReminderRecord:
    id: int
    message: str
    due_at: str


class SemanticMemoryEngine:
    """SQLite semantic memory + contradiction-aware upsert + episodic session archive."""

    def __init__(self, db_path: Path, genai_client: Any = None) -> None:
        self.db_path = Path(db_path).expanduser().resolve()
        self.db_path.parent.mkdir(parents=True, exist_ok=True)
        self._client = genai_client
        self._summary_model = (
            os.getenv("DARKSHADOW_CASUAL_MODEL", "gemini-3.1-flash-lite").strip()
            or "gemini-3.1-flash-lite"
        )
        self._init_db()

    def _connect(self) -> sqlite3.Connection:
        conn = sqlite3.connect(self.db_path, timeout=5.0)
        conn.row_factory = sqlite3.Row
        return conn

    def _init_db(self) -> None:
        with self._connect() as conn:
            conn.executescript(
                """
                CREATE TABLE IF NOT EXISTS memories (
                    id INTEGER PRIMARY KEY AUTOINCREMENT,
                    kind TEXT NOT NULL CHECK (kind IN ('preference', 'note')),
                    content TEXT NOT NULL,
                    embedding BLOB,
                    created_at TEXT NOT NULL
                );
                CREATE TABLE IF NOT EXISTS reminders (
                    id INTEGER PRIMARY KEY AUTOINCREMENT,
                    message TEXT NOT NULL,
                    due_at_iso TEXT NOT NULL,
                    delivered_at TEXT,
                    created_at TEXT NOT NULL
                );
                CREATE TABLE IF NOT EXISTS episodes (
                    id INTEGER PRIMARY KEY AUTOINCREMENT,
                    session_date TEXT NOT NULL,
                    summary TEXT NOT NULL,
                    embedding BLOB,
                    created_at TEXT NOT NULL
                );
                """
            )

    def _embed(self, text: str) -> bytes | None:
        if self._client is None:
            return None
        try:
            resp = self._client.models.embed_content(
                model="gemini-embedding-2",
                contents=text,
            )
            values = resp.embeddings[0].values
            return np.asarray(values, dtype=np.float32).tobytes()
        except Exception:
            return None

    @staticmethod
    def _cosine(a_blob: bytes | None, b_vec: np.ndarray | None) -> float:
        if not a_blob or b_vec is None:
            return 0.0
        a_vec = np.frombuffer(a_blob, dtype=np.float32)
        if a_vec.shape != b_vec.shape:
            return 0.0
        denom = float(np.linalg.norm(a_vec) * np.linalg.norm(b_vec))
        return float(np.dot(a_vec, b_vec) / denom) if denom > 1e-8 else 0.0

    @staticmethod
    def _tokens(text: str) -> set[str]:
        return set(re.findall(r"[a-z0-9\u0900-\u097f]{2,}", text.lower()))

    def _find_conflicting_preference(
        self, conn: sqlite3.Connection, content: str, emb: bytes | None
    ) -> int | None:
        """Find an existing preference that is duplicate or directly superseded by the new one."""
        new_norm = re.sub(r"\s+", " ", content.lower().strip())
        new_vec = np.frombuffer(emb, dtype=np.float32) if emb else None
        new_toks = self._tokens(content)
        neg_words = {"nahi", "not", "never", "mat", "hate", "dislike", "bilkul", "no"}
        new_has_neg = bool(new_toks & neg_words)

        best_id: int | None = None
        best_score = 0.0

        rows = conn.execute(
            "SELECT id, content, embedding FROM memories WHERE kind = 'preference'"
        ).fetchall()
        for row in rows:
            old_text = str(row["content"])
            old_norm = re.sub(r"\s+", " ", old_text.lower().strip())
            if old_norm == new_norm:
                return int(row["id"])

            sim = self._cosine(row["embedding"], new_vec)
            old_toks = self._tokens(old_text)
            shared_subject = (new_toks - neg_words) & (old_toks - neg_words)
            old_has_neg = bool(old_toks & neg_words)

            # High semantic overlap OR direct polarity flip on the same subject (e.g. chai pasand vs chai nahi pasand)
            is_polarity_flip = (new_has_neg != old_has_neg) and len(shared_subject) >= 2
            if (sim >= 0.82 or is_polarity_flip) and (sim + len(shared_subject) * 0.1) > best_score:
                best_score = sim + len(shared_subject) * 0.1
                best_id = int(row["id"])

        return best_id

    def save_memory(self, kind: str, content: str) -> tuple[bool, str]:
        cleaned = content.strip()
        if not cleaned:
            return False, "Empty memory save nahi ki ja sakti."
        if contains_sensitive_data(cleaned):
            return (
                False,
                "Privacy shield: API keys, passwords, OTPs aur payment data save karna blocked hai.",
            )

        norm_kind = "note" if kind.lower() == "note" else "preference"
        emb = self._embed(cleaned)
        now = datetime.now(timezone.utc).isoformat()

        with self._connect() as conn:
            if norm_kind == "preference":
                existing_id = self._find_conflicting_preference(conn, cleaned, emb)
                if existing_id is not None:
                    conn.execute(
                        "UPDATE memories SET content = ?, embedding = ?, created_at = ? WHERE id = ?",
                        (cleaned, emb, now, existing_id),
                    )
                    return (
                        True,
                        f"Captain, preference #{existing_id} update kar di gayi (purani conflicting entry replace ho gayi).",
                    )

            cur = conn.execute(
                "INSERT INTO memories (kind, content, embedding, created_at) VALUES (?, ?, ?, ?)",
                (norm_kind, cleaned, emb, now),
            )
            item_id = cur.lastrowid
        return True, f"Captain, {norm_kind} #{item_id} semantic memory mein save ho gaya."

    def recall_relevant(self, query: str, limit: int = 4) -> list[str]:
        q_blob = self._embed(query)
        q_vec = np.frombuffer(q_blob, dtype=np.float32) if q_blob else None
        q_words = self._tokens(query)

        scored: list[tuple[float, str]] = []
        with self._connect() as conn:
            for row in conn.execute("SELECT kind, content, embedding FROM memories").fetchall():
                sim = self._cosine(row["embedding"], q_vec)
                words = self._tokens(str(row["content"]))
                kw = len(q_words & words) * 0.25
                score = sim + kw
                if score >= 0.20 or kw > 0:
                    scored.append((score, f"[{row['kind']}] {row['content']}"))

            for row in conn.execute(
                "SELECT session_date, summary, embedding FROM episodes ORDER BY id DESC LIMIT 20"
            ).fetchall():
                sim = self._cosine(row["embedding"], q_vec)
                words = self._tokens(str(row["summary"]))
                kw = len(q_words & words) * 0.15
                score = sim + kw
                if score >= 0.35:
                    scored.append((score, f"[past session {row['session_date']}] {row['summary']}"))

        scored.sort(key=lambda x: x[0], reverse=True)
        return [item for _, item in scored[:limit]]

    def list_recent(self, limit: int = 10) -> list[str]:
        with self._connect() as conn:
            rows = conn.execute(
                "SELECT id, kind, content FROM memories ORDER BY id DESC LIMIT ?",
                (limit,),
            ).fetchall()
        return [f"#{r['id']} [{r['kind']}] {r['content']}" for r in rows]

    def archive_episode(self, turns: list[tuple[str, str]]) -> None:
        if len(turns) < 2:
            return
        transcript = "\n".join(f"Captain: {u}\nDarkShadow: {a}" for u, a in turns[-10:])
        summary = f"Discussed: {turns[-1][0][:120]}"
        if self._client is not None:
            try:
                resp = self._client.models.generate_content(
                    model=self._summary_model,
                    contents=(
                        "Summarize this conversation in 1-2 short factual sentences for long-term "
                        "memory (include any new preferences or tasks):\n" + transcript
                    ),
                )
                if resp.text and resp.text.strip():
                    summary = resp.text.strip()[:300]
            except Exception:
                pass
        if contains_sensitive_data(summary):
            return
        now = datetime.now(timezone.utc)
        emb = self._embed(summary)
        with self._connect() as conn:
            conn.execute(
                "INSERT INTO episodes (session_date, summary, embedding, created_at) VALUES (?, ?, ?, ?)",
                (now.strftime("%Y-%m-%d"), summary, emb, now.isoformat()),
            )

    def add_reminder(self, message: str, due_at: datetime) -> tuple[bool, str]:
        cleaned = message.strip()
        if not cleaned:
            return False, "Reminder message empty nahi ho sakta."
        if contains_sensitive_data(cleaned):
            return False, "Privacy shield: Sensitive data reminder mein blocked hai."
        now = datetime.now(timezone.utc).isoformat()
        if due_at.tzinfo is None:
            due_utc = due_at.astimezone(timezone.utc).isoformat()
        else:
            due_utc = due_at.astimezone(timezone.utc).isoformat()
        with self._connect() as conn:
            cur = conn.execute(
                "INSERT INTO reminders (message, due_at_iso, created_at) VALUES (?, ?, ?)",
                (cleaned, due_utc, now),
            )
            rid = cur.lastrowid
        local_disp = due_at.astimezone().strftime("%d %b %I:%M %p")
        return True, f"Captain, reminder #{rid} set ho gaya ({local_disp}): {cleaned}"

    def pop_due_reminders(self) -> list[ReminderRecord]:
        now_iso = datetime.now(timezone.utc).isoformat()
        due: list[ReminderRecord] = []
        with self._connect() as conn:
            rows = conn.execute(
                "SELECT id, message, due_at_iso FROM reminders WHERE delivered_at IS NULL AND due_at_iso <= ?",
                (now_iso,),
            ).fetchall()
            for row in rows:
                conn.execute(
                    "UPDATE reminders SET delivered_at = ? WHERE id = ?",
                    (now_iso, row["id"]),
                )
                due.append(
                    ReminderRecord(
                        id=int(row["id"]),
                        message=str(row["message"]),
                        due_at=str(row["due_at_iso"]),
                    )
                )
        return due

"""

=====================================================================================
Combines the two gold-standard Python AI engineering frameworks:
  1. Pydantic v2 / Pydantic-AI (`pydantic`, `pydantic_ai`):

def _call_gemini_structured(prompt: str, schema_cls: type[BaseModel]) -> BaseModel:
    """
    Uses `pydantic_ai.Agent` when installed, or native `google-genai` structured
    JSON output validated strictly through Pydantic v2 (`schema_cls.model_validate_json`).
    Automatically falls back from `gemini-4-hydrogen` to `gemini-3.1-flash-lite` on 503/429.
    """
    api_key = os.getenv("GEMINI_API_KEY", "").strip()
    if not api_key:
        raise RuntimeError("GEMINI_API_KEY not found in .env")

    primary = os.getenv("DARKSHADOW_REASONING_MODEL", "gemini-4-hydrogen")
    fallback = os.getenv("DARKSHADOW_FALLBACK_MODEL", "gemini-3.1-flash-lite")

    # Path A: Try Pydantic-AI native typed agent if installed
    try:
        from pydantic_ai import Agent as PydanticAgent  # type: ignore

        os.environ.setdefault("GEMINI_API_KEY", api_key)
        for model_id in (primary, fallback):
            try:
                agent = PydanticAgent(
                    f"google-gla:{model_id}",
                    output_type=schema_cls,
                    retries=2,
                )
                result = agent.run_sync(prompt)
                if isinstance(result.output, schema_cls):
                    return result.output
            except Exception:
                continue
    except Exception:
        pass

    # Path B: Direct `google-genai` structured schema validation via Pydantic v2
    from google import genai  # type: ignore
    from google.genai import types  # type: ignore

    client = genai.Client(api_key=api_key)
    last_err: Exception | None = None

    for model_id in (primary, fallback):
        for attempt in range(2):
            try:
                resp = client.models.generate_content(
                    model=model_id,
                    contents=prompt,
                    config=types.GenerateContentConfig(
                        response_mime_type="application/json",
                        response_schema=schema_cls,
                        temperature=0.15,
                    ),
                )
                raw = (resp.text or "").strip()
                return schema_cls.model_validate_json(raw)
            except ValidationError as verr:
                # Self-healing validation retry: feed exact Pydantic error back to Gemini
                prompt = (
                    f"{prompt}\n\nPREVIOUS OUTPUT FAILED PYDANTIC VALIDATION:\n{verr}\n"
                    "Return ONLY valid JSON matching the exact schema."
                )
                last_err = verr
            except Exception as exc:
                last_err = exc
                if any(tok in str(exc) for tok in ("503", "429", "UNAVAILABLE", "RESOURCE_EXHAUSTED")):
                    break
                continue

    raise RuntimeError(f"Structured reasoning failed: {last_err}")

                return LIVING_MEMORY.core_memory_append(v.block, v.content)
            if v.mode == "core_replace":
                return LIVING_MEMORY.core_memory_replace(v.block, v.old_text, v.content)
            return LIVING_MEMORY.remember(v.content)

        if tool == "memory_recall":
            return LIVING_MEMORY.recall(str(args.get("query", "")))

        if tool == "final_answer":
            return {"ok": True, "message": str(args.get("answer", "Completed."))}

        return {"ok": False, "message": f"Unknown tool '{tool}'."}

    except ValidationError as verr:
        return {"ok": False, "validation_error": str(verr), "message": f"Invalid arguments: {verr}"}
    except Exception as exc:
        log_error("reasoning_engine", f"Step '{tool}' error: {exc}")
        return {"ok": False, "message": f"Execution error: {exc}"}


# ---------------------------------------------------------------------------
# 5. LangGraph Cyclic Autonomous Engine (with Human-in-the-Loop interrupt)
# ---------------------------------------------------------------------------

class DarkShadowReasoningEngine:
    """
    Cyclic `Plan -> Safety Gate (HITL) -> Execute -> Verify/Reflect -> Synthesize`
    reasoning engine built on LangGraph + Pydantic-AI.
    """

    def __init__(self) -> None:
        self._langgraph_available = False
        self._graph: Any = None
        self._checkpointer: Any = None
        self._build_langgraph()

    def _build_langgraph(self) -> None:
        try:
            from langgraph.checkpoint.memory import MemorySaver  # type: ignore
            from langgraph.graph import END, START, StateGraph  # type: ignore
            from langgraph.types import interrupt  # type: ignore
        except Exception:
            self._langgraph_available = False
            return

        def plan_node(state: AgentGraphState) -> dict[str, Any]:
            planner_prompt = (
                "You are DarkShadow's Master Planner for Windows 11.\n"
                f"{state['context']}\n\n"
                f"USER GOAL: {state['goal']}\n\n"
                "Create a minimal, deterministic step-by-step plan (1 to 6 steps).\n"
                "Available tools:\n"
                "- web_autonomous: {goal, start_url?, max_steps?} (full multi-step Edge browser-use agent)\n"
                "- web_a11y_snapshot: {url?, max_nodes?} (fast Playwright Accessibility Tree read)\n"
                "- windows_uia_inspect: {window_title?} (read native Windows app UI controls)\n"
                "- windows_uia_click: {index?, name?, window_title?} (click native Windows control)\n"
                "- windows_uia_type: {text, index?, name?, press_enter?, window_title?}\n"
                "- windows_app_open: {app_name}\n"
                "- windows_system: {action: 'status'|'volume'|'brightness', level?}\n"
                "- mcp_hub_call: {server_name, tool_name, arguments}\n"
                "- memory_update: {mode: 'remember'|'core_append'|'core_replace', content, block?}\n"
                "- memory_recall: {query}\n"
                "- final_answer: {answer}\n"
                "Mark requires_confirmation=true for any risky/delete/send action."
            )
            plan_obj = _call_gemini_structured(planner_prompt, ExecutionPlan)
            assert isinstance(plan_obj, ExecutionPlan)
            return {
                "plan": [s.model_dump() for s in plan_obj.steps],
                "current_step_idx": 0,
                "retries_on_step": 0,
                "step_logs": [],
                "awaiting_approval": False,
                "approved": True,
            }

        def safety_gate_node(state: AgentGraphState) -> dict[str, Any]:
            idx = state["current_step_idx"]
            if idx >= len(state["plan"]):
                return {}
            step = PlanStep.model_validate(state["plan"][idx])
            serialized = json.dumps(step.args, ensure_ascii=False)

            shield_reason = check_hard_shields(serialized)
            if shield_reason:
                return {
                    "final_answer": f"Captain, Safety Shield ne risky step block kar diya: {shield_reason}",
                    "current_step_idx": len(state["plan"]),
                }

            needs_hitl = step.requires_confirmation or bool(
                _SENSITIVE_STEP_RE.search(f"{step.tool} {serialized}")
            )
            if needs_hitl:
                # Official LangGraph Human-in-the-Loop interrupt()!
                decision = interrupt(
                    {
                        "type": "captain_approval_required",
                        "step_id": step.step_id,
                        "tool": step.tool,
                        "args": step.args,
                        "message": f"Captain, kya main step {step.step_id} ({step.tool}) execute karoon?",
                    }
                )
                if not bool(decision):
                    return {
                        "approved": False,
                        "final_answer": f"Captain ke instruction par step {step.step_id} cancel kar diya gaya.",
                        "current_step_idx": len(state["plan"]),
                    }
            return {"approved": True}

        def execute_step_node(state: AgentGraphState) -> dict[str, Any]:
            idx = state["current_step_idx"]
            if idx >= len(state["plan"]):
                return {}
            step = PlanStep.model_validate(state["plan"][idx])
            res = _execute_validated_step(step)
            logs = list(state["step_logs"])
            logs.append({"step": step.step_id, "tool": step.tool, "args": step.args, "result": res})
            return {"step_logs": logs}

        def reflect_node(state: AgentGraphState) -> dict[str, Any]:
            idx = state["current_step_idx"]
            if idx >= len(state["plan"]) or not state["step_logs"]:
                return {"current_step_idx": len(state["plan"])}

            last_log = state["step_logs"][-1]
            res = last_log["result"]

            # Fast path: if step succeeded, advance directly (saves tokens!)
            if res.get("ok"):
                return {"current_step_idx": idx + 1, "retries_on_step": 0}

            # Self-correcting retry path (max 2 retries per step)
            if state["retries_on_step"] < 2:
                reflect_prompt = (
                    f"Goal: {state['goal']}\n"
                    f"Failed Step: {json.dumps(last_log, ensure_ascii=False)[:1500]}\n"
                    "Decide whether to retry_step with corrected_args, continue, or abort."
                )
                try:
                    decision = _call_gemini_structured(reflect_prompt, ReflectionDecision)
                    assert isinstance(decision, ReflectionDecision)
                    if decision.status == "retry_step" and decision.corrected_args:
                        updated_plan = list(state["plan"])
                        updated_plan[idx] = {**updated_plan[idx], "args": decision.corrected_args}
                        return {
                            "plan": updated_plan,
                            "retries_on_step": state["retries_on_step"] + 1,
                        }
                    if decision.status == "abort":
                        return {
                            "current_step_idx": len(state["plan"]),
                            "final_answer": f"Captain, step fail hone par task safely rok diya: {decision.reason}",
                        }
                except Exception:
                    pass

            return {"current_step_idx": idx + 1, "retries_on_step": 0}

        def synthesize_node(state: AgentGraphState) -> dict[str, Any]:
            if state.get("final_answer"):
                return {}
            summary_lines = [
                f"Step {l['step']} ({l['tool']}): {'OK' if l['result'].get('ok') else 'FAIL'} — {str(l['result'].get('message', ''))[:240]}"
                for l in state["step_logs"]
            ]
            answer = "Captain, autonomous task complete:\n" + "\n".join(summary_lines)
            return {"final_answer": answer}

        def route_after_reflect(state: AgentGraphState) -> str:
            if state.get("final_answer") or state["current_step_idx"] >= len(state["plan"]):
                return "synthesize"
            return "safety_gate"

        builder = StateGraph(AgentGraphState)
        builder.add_node("plan", plan_node)
        builder.add_node("safety_gate", safety_gate_node)
        builder.add_node("execute_step", execute_step_node)
        builder.add_node("reflect", reflect_node)
        builder.add_node("synthesize", synthesize_node)

        builder.add_edge(START, "plan")
        builder.add_edge("plan", "safety_gate")
        builder.add_edge("safety_gate", "execute_step")
        builder.add_edge("execute_step", "reflect")
        builder.add_conditional_edges(
            "reflect",
            route_after_reflect,
            {"safety_gate": "safety_gate", "synthesize": "synthesize"},
        )
        builder.add_edge("synthesize", END)

        self._checkpointer = MemorySaver()
        self._graph = builder.compile(checkpointer=self._checkpointer)
        self._langgraph_available = True

    def status(self) -> dict[str, Any]:
        pydantic_ai_ok = False
        try:
            import pydantic_ai  # noqa: F401
            pydantic_ai_ok = True
        except Exception:
            pass
        return {
            "ok": True,
            "langgraph_active": self._langgraph_available,
            "pydantic_ai_active": pydantic_ai_ok,
            "pydantic_v2_active": True,
        }

    def run_goal(self, goal: str, thread_id: str | None = None) -> dict[str, Any]:
        """
        Runs a complex multi-step goal through LangGraph + Pydantic-AI.
        If Human-in-the-Loop interrupt triggers, returns `needs_approval=True` and `thread_id`
        so Captain can resume with `resume_approval(thread_id, approved=True)`.
        """
        context = LIVING_MEMORY.build_system_context_block(current_query=goal)
        tid = thread_id or f"ds-{uuid.uuid4().hex[:8]}"

        initial_state: AgentGraphState = {
            "goal": goal,
            "context": context,
            "plan": [],
            "current_step_idx": 0,
            "retries_on_step": 0,
            "step_logs": [],
            "awaiting_approval": False,
            "approval_prompt": "",
            "approved": True,
            "final_answer": "",
        }

        if self._langgraph_available and self._graph is not None:
            try:
                config = {"configurable": {"thread_id": tid}}
                result = self._graph.invoke(initial_state, config=config)
                snapshot = self._graph.get_state(config)
                if getattr(snapshot, "next", None):
                    # Graph paused at Human-in-the-Loop interrupt()
                    interrupts = []
                    for task in getattr(snapshot, "tasks", ()) or ():
                        for intr in getattr(task, "interrupts", ()) or ():
                            interrupts.append(intr.value)
                    return {
                        "ok": True,
                        "engine": "langgraph",
                        "thread_id": tid,
                        "needs_approval": True,
                        "approval_details": interrupts,
                        "message": (
                            interrupts[0].get("message")
                            if interrupts and isinstance(interrupts[0], dict)
                            else "Captain, risky step ke liye aapki approval chahiye."
                        ),
                    }
                return {
                    "ok": True,
                    "engine": "langgraph",
                    "thread_id": tid,
                    "needs_approval": False,
                    "steps_executed": len(result.get("step_logs", [])),
                    "logs": result.get("step_logs", []),
                    "message": result.get("final_answer", "Task completed."),
                }
            except Exception as exc:
                log_error("reasoning_engine", f"LangGraph run error: {exc}")
                return {"ok": False, "engine": "langgraph", "message": f"Reasoning error: {exc}"}

        # Pure-Python fallback cyclic loop if `langgraph` is not installed yet
        try:
            plan_obj = _call_gemini_structured(
                f"Create a 1-5 step plan for Windows 11 goal: {goal}\nContext:\n{context}",
                ExecutionPlan,
            )
            assert isinstance(plan_obj, ExecutionPlan)
            logs: list[dict[str, Any]] = []
            for step in plan_obj.steps:
                if step.requires_confirmation:
                    return {
                        "ok": False,
                        "engine": "python_fallback",
                        "needs_approval": True,
                        "message": f"Captain, risky step ({step.tool}) paused. Confirm explicitly before proceeding.",
                    }
                res = _execute_validated_step(step)
                logs.append({"step": step.step_id, "tool": step.tool, "result": res})
            return {
                "ok": True,
                "engine": "python_fallback",
                "steps_executed": len(logs),
                "logs": logs,
                "message": "Captain, task complete: " + "; ".join(
                    str(l["result"].get("message", ""))[:180] for l in logs
                ),
            }
        except Exception as exc:
            log_error("reasoning_engine", f"Fallback planner error: {exc}")
            return {"ok": False, "message": f"Reasoning failed: {exc}"}

    def resume_approval(self, thread_id: str, approved: bool) -> dict[str, Any]:
        """Resumes a LangGraph Human-in-the-Loop interrupted workflow after Captain replies."""
        if not self._langgraph_available or self._graph is None:
            return {"ok": False, "message": "LangGraph checkpointer not active."}
        try:
            from langgraph.types import Command  # type: ignore

            config = {"configurable": {"thread_id": thread_id}}
            result = self._graph.invoke(Command(resume=bool(approved)), config=config)
            return {
                "ok": True,
                "engine": "langgraph",
                "thread_id": thread_id,
                "approved": bool(approved),
                "logs": result.get("step_logs", []),
                "message": result.get("final_answer", "Resumed workflow complete."),
            }
        except Exception as exc:
            log_error("reasoning_engine", f"resume_approval failed: {exc}")
            return {"ok": False, "message": f"Resume failed: {exc}"}


REASONING_ENGINE = DarkShadowReasoningEngine()

"""
DarkShadow Antigravity Error Tracker
====================================
Captures runtime errors, tool failures, model fallbacks, and safety blocks
to .darkshadow/antigravity_error_log.jsonl for inspection and repair.
"""

from __future__ import annotations

import json
import os
import traceback
from datetime import datetime, timezone
from pathlib import Path
from typing import Any

LOG_FILE = Path(__file__).resolve().parent / ".darkshadow" / "antigravity_error_log.jsonl"


def log_error(
    category: str,
    error_message: str,
    command_or_context: str = "",
    exception: Exception | None = None,
    suggested_fix: str = "",
    status: str = "PENDING_FIX",
) -> dict[str, Any]:
    """Record a structured error event to the central error log."""
    try:
        LOG_FILE.parent.mkdir(parents=True, exist_ok=True)
        now = datetime.now(timezone.utc).isoformat()
        err_id = f"ERR-{datetime.now().strftime('%Y%m%d')}-{os.urandom(2).hex().upper()}"

        tb_str = ""
        if exception is not None:
            tb_str = "".join(
                traceback.format_exception(type(exception), exception, exception.__traceback__)
            )

        record = {
            "id": err_id,
            "timestamp": now,
            "status": status,
            "category": category,
            "context": command_or_context,
            "error_message": str(error_message),
            "stack_trace": tb_str[-800:] if tb_str else "",
            "suggested_fix": suggested_fix,
        }

        with open(LOG_FILE, "a", encoding="utf-8") as f:
            f.write(json.dumps(record, ensure_ascii=False) + "\n")

        return record
    except Exception as exc:
        print(f"[ErrorTracker] Failed to record error: {exc}")
        return {}


def list_pending_errors(limit: int = 50) -> list[dict[str, Any]]:
    """Return recent pending errors."""
    if not LOG_FILE.exists():
        return []
    records: list[dict[str, Any]] = []
    try:
        with open(LOG_FILE, "r", encoding="utf-8") as f:
            for line in f:
                if line.strip():
                    rec = json.loads(line)
                    if rec.get("status") == "PENDING_FIX":
                        records.append(rec)
    except Exception:
        pass
    return records[-limit:]
